Typing speed test

Question: Write a JavaScript program to typing speed test.
▶ Live Output
Working example
JavaScript Code
const samplePassages = [
      "JavaScript is a high-level, just-in-time compiled language conforming to the ECMAScript specification.",
      "The Document Object Model represents the web page as nodes that can be modified dynamically with code.",
      "Asynchronous programming with promises and async await patterns makes web apps responsive and scalable."
    ];

    let currentPassage = "";
    let timeLeft = 60;
    let timer = null;
    let started = false;
    let totalErrors = 0;

    const quoteEl = document.getElementById("quoteDisplay");
    const inputEl = document.getElementById("typingInput");
    const wpmEl = document.getElementById("statWpm");
    const accEl = document.getElementById("statAccuracy");
    const timeEl = document.getElementById("statTime");
    const errEl = document.getElementById("statErrors");

    function setupPassage() {
      currentPassage = samplePassages[Math.floor(Math.random() * samplePassages.length)];
      quoteEl.innerHTML = "";
      currentPassage.split("").forEach((char, i) => {
        const span = document.createElement("span");
        span.textContent = char;
        if (i === 0) span.className = "char-current";
        quoteEl.appendChild(span);
      });
    }

    function processInput() {
      if (!started) {
        started = true;
        startTimer();
      }

      const inputVal = inputEl.value;
      const spans = quoteEl.querySelectorAll("span");
      let errors = 0;

      spans.forEach((span, i) => {
        const char = inputVal[i];
        span.className = "";

        if (char == null) {
          if (i === inputVal.length) span.className = "char-current";
        } else if (char === span.textContent) {
          span.className = "char-correct";
        } else {
          span.className = "char-incorrect";
          errors++;
        }
      });

      totalErrors = errors;
      errEl.textContent = errors;

      const typedCount = inputVal.length;
      const elapsedMins = (60 - timeLeft) / 60 || 0.01;
      const wpm = Math.max(0, Math.round(((typedCount / 5) - errors) / elapsedMins));
      wpmEl.textContent = wpm;

      const accuracy = typedCount > 0 ? Math.max(0, Math.round(((typedCount - errors) / typedCount) * 100)) : 100;
      accEl.textContent = `${accuracy}%`;

      if (inputVal.length === currentPassage.length) {
        clearInterval(timer);
        inputEl.disabled = true;
      }
    }

    function startTimer() {
      timer = setInterval(() => {
        timeLeft--;
        timeEl.textContent = `${timeLeft}s`;
        if (timeLeft <= 0) {
          clearInterval(timer);
          inputEl.disabled = true;
        }
      }, 1000);
    }

    function resetTypingTest() {
      clearInterval(timer);
      timer = null;
      started = false;
      timeLeft = 60;
      inputEl.value = "";
      inputEl.disabled = false;
      timeEl.textContent = "60s";
      wpmEl.textContent = "0";
      accEl.textContent = "100%";
      errEl.textContent = "0";
      setupPassage();
    }

    setupPassage();
