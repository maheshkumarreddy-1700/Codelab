Countdown timer

Question: Write a JavaScript program to countdown timer.
▶ Live Output
Working example
JavaScript Code
let totalSeconds = 150;
    let timer = null;

    const readout = document.getElementById("readout");
    const stateBadge = document.getElementById("stateBadge");
    const btnStart = document.getElementById("btnStart");
    const btnPause = document.getElementById("btnPause");

    function updateDisplay() {
      const m = String(Math.floor(totalSeconds / 60)).padStart(2, '0');
      const s = String(totalSeconds % 60).padStart(2, '0');
      readout.textContent = `${m}:${s}`;
    }

    function startTimer() {
      if (timer) return;

      if (totalSeconds === 0) {
        const mins = parseInt(document.getElementById("inputMins").value) || 0;
        const secs = parseInt(document.getElementById("inputSecs").value) || 0;
        totalSeconds = (mins * 60) + secs;
      }

      readout.classList.remove("alarm-active");
      stateBadge.textContent = "Running";
      stateBadge.className = "badge badge-primary";
      btnStart.disabled = true;
      btnPause.disabled = false;

      timer = setInterval(() => {
        if (totalSeconds <= 0) {
          clearInterval(timer);
          timer = null;
          readout.classList.add("alarm-active");
          stateBadge.textContent = "Time's Up!";
          stateBadge.className = "badge badge-warning";
          btnStart.disabled = false;
          btnPause.disabled = true;
          return;
        }
        totalSeconds--;
        updateDisplay();
      }, 1000);
    }

    function pauseTimer() {
      clearInterval(timer);
      timer = null;
      stateBadge.textContent = "Paused";
      stateBadge.className = "badge badge-secondary";
      btnStart.disabled = false;
      btnPause.disabled = true;
    }

    function resetCountdown() {
      clearInterval(timer);
      timer = null;
      const mins = parseInt(document.getElementById("inputMins").value) || 0;
      const secs = parseInt(document.getElementById("inputSecs").value) || 0;
      totalSeconds = (mins * 60) + secs;
      readout.classList.remove("alarm-active");
      updateDisplay();
      stateBadge.textContent = "Idle";
      stateBadge.className = "badge badge-success";
      btnStart.disabled = false;
      btnPause.disabled = true;
    }

    document.getElementById("inputMins").addEventListener("change", resetCountdown);
    document.getElementById("inputSecs").addEventListener("change", resetCountdown);
    resetCountdown();
