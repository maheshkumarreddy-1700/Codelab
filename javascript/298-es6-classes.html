Es6 classes

Question: Write a JavaScript program to es6 classes.
▶ Live Output
Working example
JavaScript Code
class Person {
      constructor(name) { this.name = name; }
      introduce() { return `Member: ${this.name}`; }
    }
    class Student extends Person {
      constructor(name, department, gpa) {
        super(name);
        this.department = department;
        this.gpa = gpa;
      }
      getHonors() {
        return this.gpa >= 8.5 ? "First Class with Distinction" : "First Class";
      }
    }

    function instantiateStudent() {
      const name = document.getElementById('stud-name').value;
      const dept = document.getElementById('stud-branch').value;
      const gpa = parseFloat(document.getElementById('stud-gpa').value);

      const s = new Student(name, dept, gpa);
      document.getElementById('res-oop').innerHTML = `
        <strong>Instance Created:</strong> new Student("${s.name}", "${s.department}", ${s.gpa})<br>
        <strong>Inherited Method (Person.introduce):</strong> "${s.introduce()}"<br>
        <strong>Subclass Method (Student.getHonors):</strong> <span style="color: #16a34a; font-weight: 700;">${s.getHonors()}</span><br>
        <strong>Instance Check:</strong> (s instanceof Student: ${s instanceof Student}) &amp;&amp; (s instanceof Person: ${s instanceof Person})
      `;
    }
    instantiateStudent();
