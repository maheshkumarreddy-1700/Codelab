Question: Write a JavaScript program to persistent todo.

let todos = JSON.parse(localStorage.getItem("lab_persistent_todos") || "null");

    if (!todos) {
      todos = [
        { id: 1, text: "Review HTML5 semantic structure", done: true },
        { id: 2, text: "Practice CSS Flexbox and Grid layouts", done: true },
        { id: 3, text: "Complete JavaScript LocalStorage lab experiment", done: false }
      ];
      localStorage.setItem("lab_persistent_todos", JSON.stringify(todos));
    }

    const listEl = document.getElementById("todoList");
    const countEl = document.getElementById("todoCount");

    function saveAndRender() {
      localStorage.setItem("lab_persistent_todos", JSON.stringify(todos));
      render();
    }

    function render() {
      listEl.innerHTML = "";
      countEl.textContent = todos.length;

      if (todos.length === 0) {
        listEl.innerHTML = `<li style="text-align:center; color:#64748b; padding:1.5rem;">All tasks completed! Add a new task above.</li>`;
        return;
      }

      todos.forEach(t => {
        const li = document.createElement("li");
        li.className = `todo-item ${t.done ? 'done' : ''}`;
        li.innerHTML = `
          <div style="display:flex; align-items:center;">
            <input type="checkbox" class="check-toggle" ${t.done ? 'checked' : ''} onchange="toggleTodo(${t.id})">
            <span>${t.text}</span>
          </div>
          <button class="btn-del" onclick="deleteTodo(${t.id})" title="Delete">&times;</button>
        `;
        listEl.appendChild(li);
      });
    }

    function addTask() {
      const input = document.getElementById("taskInput");
      const text = input.value.trim();
      if (!text) return;

      todos.push({ id: Date.now(), text, done: false });
      input.value = "";
      saveAndRender();
    }

    function toggleTodo(id) {
      const item = todos.find(t => t.id === id);
      if (item) item.done = !item.done;
      saveAndRender();
    }

    function deleteTodo(id) {
      todos = todos.filter(t => t.id !== id);
      saveAndRender();
    }

    function clearCompleted() {
      todos = todos.filter(t => !t.done);
      saveAndRender();
    }

    render();
