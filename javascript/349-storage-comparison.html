Storage comparison

Question: Write a JavaScript program to storage comparison.
▶ Live Output
Working example
JavaScript Code
function saveLocal(val) {
      localStorage.setItem("probe_local", val);
      document.getElementById("valLocal").textContent = val;
    }

    function saveSession(val) {
      sessionStorage.setItem("probe_session", val);
      document.getElementById("valSession").textContent = val;
    }

    // Restore on load
    const loc = localStorage.getItem("probe_local") || "";
    const ses = sessionStorage.getItem("probe_session") || "";

    document.getElementById("testLocal").value = loc;
    document.getElementById("valLocal").textContent = loc || "(none)";

    document.getElementById("testSession").value = ses;
    document.getElementById("valSession").textContent = ses || "(none)";
