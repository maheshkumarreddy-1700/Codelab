function calculateAll() {
      const a = parseFloat(document.getElementById('num1').value) || 0;
      const b = parseFloat(document.getElementById('num2').value) || 0;

      const ops = [
        { name: "Addition", symbol: "+", expr: `${a} + ${b}`, res: a + b },
        { name: "Subtraction", symbol: "-", expr: `${a} - ${b}`, res: a - b },
        { name: "Multiplication", symbol: "*", expr: `${a} * ${b}`, res: a * b },
        { name: "Division", symbol: "/", expr: `${a} / ${b}`, res: b !== 0 ? (a / b).toFixed(4) : "Undefined (Div by 0)" },
        { name: "Modulus (Remainder)", symbol: "%", expr: `${a} % ${b}`, res: b !== 0 ? (a % b) : "Undefined" },
        { name: "Exponentiation", symbol: "**", expr: `${a} ** ${b}`, res: a ** b }
      ];

      const tbody = document.getElementById('arithmetic-results');
      tbody.innerHTML = ops.map(op => `
        <tr>
          <td style="padding: 0.6rem 0.75rem; border: 1px solid var(--border); font-weight: 600;">${op.name}</td>
          <td style="padding: 0.6rem 0.75rem; border: 1px solid var(--border); font-family: var(--font-mono);">${op.symbol}</td>
          <td style="padding: 0.6rem 0.75rem; border: 1px solid var(--border); font-family: var(--font-mono); color: #64748b;">${op.expr}</td>
          <td style="padding: 0.6rem 0.75rem; border: 1px solid var(--border); font-family: var(--font-mono); font-weight: 700; color: #4f46e5;">${op.res}</td>
        </tr>
      `).join('');
    }
    calculateAll();
