Dynamic css

Question: Write a JavaScript program to dynamic css.
▶ Live Output
Working example
JavaScript Code
const card = document.getElementById("targetCard");

    function updateStyles() {
      const bg = document.getElementById("bgColor").value;
      const color = document.getElementById("textColor").value;
      const size = document.getElementById("fontSize").value;
      const radius = document.getElementById("borderRadius").value;
      const rot = document.getElementById("rotation").value;

      document.getElementById("fontSizeVal").textContent = size;
      document.getElementById("radiusVal").textContent = radius;
      document.getElementById("rotateVal").textContent = rot;

      card.style.backgroundColor = bg;
      card.style.color = color;
      card.style.fontSize = size + "px";
      card.style.borderRadius = radius + "px";
      card.style.transform = `rotate(${rot}deg)`;
    }

    function resetCardStyles() {
      document.getElementById("bgColor").value = "#3b82f6";
      document.getElementById("textColor").value = "#ffffff";
      document.getElementById("fontSize").value = 18;
      document.getElementById("borderRadius").value = 8;
      document.getElementById("rotation").value = 0;
      updateStyles();
    }
