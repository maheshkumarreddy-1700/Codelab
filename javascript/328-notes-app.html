Notes app

Question: Write a JavaScript program to notes app.

(function() {
      let notes = JSON.parse(localStorage.getItem('weblab_notes')) || [
        { id: 1, title: 'Web Lab Project', content: 'Complete all 150 experiments with interactive code viewers and responsive styling!', color: 'yellow', date: 'Sep 21, 2026' },
        { id: 2, title: 'Algorithms Revision', content: 'Review DOM tree traversal, event bubbling, and closures for technical interview.', color: 'blue', date: 'Sep 21, 2026' },
        { id: 3, title: 'Book Recommendations', content: '1. Eloquent JavaScript\n2. You Don\'t Know JS\n3. Clean Code', color: 'green', date: 'Sep 21, 2026' }
      ];

      let activeColor = 'yellow';
      const notesGrid = document.getElementById('notesGrid');
      const noteSearch = document.getElementById('noteSearch');
      const btnCreateNote = document.getElementById('btnCreateNote');

      function escapeHTML(str) {
        return str.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
      }

      function saveNotes() {
        localStorage.setItem('weblab_notes', JSON.stringify(notes));
      }

      window.selectColor = function(c) {
        activeColor = c;
        alert(`Selected ${c} color for new sticky notes.`);
      };

      window.updateNote = function(id, key, val) {
        const note = notes.find(n => n.id === id);
        if (note) {
          note[key] = val;
          saveNotes();
        }
      };

      window.deleteNote = function(id) {
        notes = notes.filter(n => n.id !== id);
        saveNotes();
        renderNotes(noteSearch.value);
      };

      function renderNotes(query = '') {
        const q = query.toLowerCase().trim();
        const filtered = notes.filter(n => 
          n.title.toLowerCase().includes(q) || 
          n.content.toLowerCase().includes(q)
        );

        if (filtered.length === 0) {
          notesGrid.innerHTML = '<div class="empty-notes-hint">No notes found. Click "+ New Note" to create one!</div>';
          return;
        }

        notesGrid.innerHTML = filtered.map(n => `
          <div class="note-card ${n.color}" data-id="${n.id}">
            <div class="note-title" contenteditable="true" onblur="window.updateNote(${n.id}, 'title', this.innerText)">${escapeHTML(n.title)}</div>
            <div class="note-body" contenteditable="true" onblur="window.updateNote(${n.id}, 'content', this.innerText)">${escapeHTML(n.content)}</div>
            <div class="note-footer">
              <span>${n.date}</span>
              <button class="note-del-btn" onclick="window.deleteNote(${n.id})" title="Delete">&times;</button>
            </div>
          </div>
        `).join('');
      }

      btnCreateNote.addEventListener('click', () => {
        const newNote = {
          id: Date.now(),
          title: 'Quick Memo',
          content: 'Click here to edit note content...',
          color: activeColor,
          date: new Date().toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
        };
        notes.unshift(newNote);
        saveNotes();
        renderNotes(noteSearch.value);
      });

      noteSearch.addEventListener('input', (e) => {
        renderNotes(e.target.value);
      });

      renderNotes();
    })();
