Login validation

Question: Write a JavaScript program to login validation.
▶ Live Output
Working example
JavaScript Code
function validateLogin(e) {
      e.preventDefault();
      const email = document.getElementById("loginEmail");
      const pwd = document.getElementById("loginPwd");
      const errEmail = document.getElementById("errLoginEmail");
      const errPwd = document.getElementById("errLoginPwd");
      const btn = document.getElementById("loginBtn");

      let valid = true;

      if (email.value.trim().length < 3) {
        email.classList.add("is-invalid");
        errEmail.style.display = "block";
        valid = false;
      } else {
        email.classList.remove("is-invalid");
        errEmail.style.display = "none";
      }

      if (pwd.value.trim() === "") {
        pwd.classList.add("is-invalid");
        errPwd.style.display = "block";
        valid = false;
      } else {
        pwd.classList.remove("is-invalid");
        errPwd.style.display = "none";
      }

      if (valid) {
        btn.textContent = "Verifying...";
        btn.disabled = true;
        setTimeout(() => {
          btn.textContent = "Sign In to Portal";
          btn.disabled = false;
          alert(`Welcome back, ${email.value}! Authentication successful.`);
        }, 800);
      }
    }
