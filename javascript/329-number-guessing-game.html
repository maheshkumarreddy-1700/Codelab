Number guessing game

Question: Write a JavaScript program to number guessing game.
▶ Live Output
Working example
JavaScript Code
let secretNumber = Math.floor(Math.random() * 100) + 1;
    let tries = 7;
    let guesses = [];

    const inp = document.getElementById("inpGuess");
    const hint = document.getElementById("hintBox");
    const badge = document.getElementById("triesBadge");
    const historyBox = document.getElementById("guessHistory");
    const btnSubmit = document.getElementById("btnSubmitGuess");

    function submitGuess() {
      const val = parseInt(inp.value);
      if (isNaN(val) || val < 1 || val > 100) {
        alert("Please enter a valid number between 1 and 100!");
        return;
      }

      guesses.push(val);
      tries--;

      if (guesses.length === 1) historyBox.innerHTML = "";
      const chip = document.createElement("span");
      chip.className = "chip";
      chip.textContent = val;
      historyBox.appendChild(chip);

      if (val === secretNumber) {
        hint.innerHTML = `🎉 <strong>BINGO!</strong> You found the secret number (${secretNumber}) in ${7 - tries} attempts!`;
        hint.style.color = "#10b981";
        hint.style.background = "#065f46";
        badge.textContent = "🏆 Victory!";
        badge.className = "badge badge-success";
        inp.disabled = true;
        btnSubmit.disabled = true;
        return;
      }

      if (tries <= 0) {
        hint.innerHTML = `💀 <strong>Game Over!</strong> The secret number was <strong>${secretNumber}</strong>.`;
        hint.style.color = "#ef4444";
        hint.style.background = "#7f1d1d";
        badge.textContent = "0 Tries (Defeat)";
        badge.className = "badge badge-warning";
        inp.disabled = true;
        btnSubmit.disabled = true;
        return;
      }

      badge.textContent = `${tries} Tries Remaining`;
      if (val < secretNumber) {
        hint.textContent = `📈 Too LOW! Try a higher number. (${val} < ?)`;
        hint.style.color = "#f59e0b";
      } else {
        hint.textContent = `📉 Too HIGH! Try a lower number. (? < ${val})`;
        hint.style.color = "#38bdf8";
      }

      inp.value = "";
      inp.focus();
    }

    function restartGame() {
      secretNumber = Math.floor(Math.random() * 100) + 1;
      tries = 7;
      guesses = [];
      inp.disabled = false;
      btnSubmit.disabled = false;
      inp.value = "";
      badge.textContent = "7 Tries Remaining";
      badge.className = "badge badge-warning";
      hint.textContent = "Make your first guess!";
      hint.style.color = "#38bdf8";
      hint.style.background = "#1e293b";
      historyBox.innerHTML = '<span style="color: #64748b; font-size: 0.8rem;">No guesses yet</span>';
      inp.focus();
    }
