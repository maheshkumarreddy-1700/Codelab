Validate name email phone number and password using javascript

Question: Write a JavaScript program to validate name email phone number and password using javascript.
▶ Live Output
Working example
JavaScript Code
function validate() {

    let name = document.getElementById("name").value.trim();
    let email = document.getElementById("email").value.trim();
    let phone = document.getElementById("phone").value.trim();
    let password = document.getElementById("password").value;

    let emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    let phonePattern = /^[0-9]{10}$/;

    if (name === "") {
        show("Enter your name.");
        return false;
    }

    if (!emailPattern.test(email)) {
        show("Enter a valid email.");
        return false;
    }

    if (!phonePattern.test(phone)) {
        show("Enter a valid 10-digit phone number.");
        return false;
    }

    if (password.length < 6) {
        show("Password must contain at least 6 characters.");
        return false;
    }

    show("Registration successful!");
    return false;
}

function show(message) {
    document.getElementById("message").textContent = message;
}
