Shopping cart app

Question: Write a JavaScript program to shopping cart app.
▶ Live Output
Working example
JavaScript Code
(function() {
      const products = [
        { id: 101, name: 'Mechanical Keyboard', price: 99.99, icon: '⌨️', category: 'Peripherals' },
        { id: 102, name: 'Ergonomic Mouse', price: 49.99, icon: '🖱️', category: 'Peripherals' },
        { id: 103, name: 'Noise-Canceling Pro', price: 149.99, icon: '🎧', category: 'Audio' },
        { id: 104, name: 'Curved 4K Display', price: 349.99, icon: '🖥️', category: 'Displays' }
      ];

      let cart = [
        { id: 101, name: 'Mechanical Keyboard', price: 99.99, icon: '⌨️', qty: 1 }
      ];
      let discountRate = 0.0;

      function renderCatalog() {
        const grid = document.getElementById('catalogGrid');
        grid.innerHTML = products.map(p => `
          <div class="product-card">
            <div class="prod-emoji">${p.icon}</div>
            <div class="prod-category">${p.category}</div>
            <div class="prod-title">${p.name}</div>
            <div class="prod-price">$${p.price.toFixed(2)}</div>
            <button class="btn-add-cart" onclick="window.addToCart(${p.id})">Add to Bag</button>
          </div>
        `).join('');
      }

      window.addToCart = function(id) {
        const existing = cart.find(item => item.id === id);
        if (existing) {
          existing.qty++;
        } else {
          const prod = products.find(p => p.id === id);
          cart.push({ ...prod, qty: 1 });
        }
        updateCart();
      };

      window.updateQty = function(id, delta) {
        const item = cart.find(i => i.id === id);
        if (!item) return;
        item.qty += delta;
        if (item.qty <= 0) {
          cart = cart.filter(i => i.id !== id);
        }
        updateCart();
      };

      function updateCart() {
        const list = document.getElementById('cartItemsList');
        const countBadge = document.getElementById('cartCountBadge');
        
        const totalItems = cart.reduce((acc, i) => acc + i.qty, 0);
        countBadge.textContent = `${totalItems} items`;

        if (cart.length === 0) {
          list.innerHTML = '<li class="empty-cart-msg">Your shopping bag is empty.</li>';
        } else {
          list.innerHTML = cart.map(i => `
            <li class="cart-item">
              <div>
                <div style="font-weight:600;">${i.name}</div>
                <div style="font-size:0.8rem; color:var(--text-secondary);">$${i.price.toFixed(2)} each</div>
              </div>
              <div class="qty-controls">
                <button class="qty-btn" onclick="window.updateQty(${i.id}, -1)">&minus;</button>
                <span style="font-weight:700; width:22px; text-align:center;">${i.qty}</span>
                <button class="qty-btn" onclick="window.updateQty(${i.id}, 1)">&plus;</button>
              </div>
            </li>
          `).join('');
        }

        const subtotal = cart.reduce((acc, i) => acc + (i.price * i.qty), 0);
        const discount = subtotal * discountRate;
        const tax = (subtotal - discount) * 0.08;
        const grandTotal = (subtotal - discount) + tax;

        document.getElementById('summarySubtotal').textContent = `$${subtotal.toFixed(2)}`;
        document.getElementById('summaryDiscount').textContent = `-$${discount.toFixed(2)}`;
        document.getElementById('summaryTax').textContent = `$${tax.toFixed(2)}`;
        document.getElementById('summaryTotal').textContent = `$${grandTotal.toFixed(2)}`;
      }

      document.getElementById('btnApplyCoupon').addEventListener('click', () => {
        const code = document.getElementById('couponCode').value.trim().toUpperCase();
        if (code === 'SAVE20') {
          discountRate = 0.20;
          document.getElementById('discountRow').style.display = 'flex';
          alert('Coupon SAVE20 applied: 20% Discount applied!');
        } else {
          alert('Invalid promo code. Please enter: SAVE20');
        }
        updateCart();
      });

      document.getElementById('btnCheckout').addEventListener('click', () => {
        if (cart.length === 0) {
          alert('Your shopping bag is empty! Add products to checkout.');
          return;
        }
        const total = document.getElementById('summaryTotal').textContent;
        alert(`Order successfully processed for ${total}!\nThank you for shopping with WebLab Studio.`);
        cart = [];
        discountRate = 0.0;
        document.getElementById('discountRow').style.display = 'none';
        document.getElementById('couponCode').value = '';
        updateCart();
      });

      renderCatalog();
      updateCart();
    })();
