Dynamic form fields

Question: Write a JavaScript program to dynamic form fields.

JavaScript Code
const container = document.getElementById("educationRows");

    function addEducationRow() {
      const row = document.createElement("div");
      row.className = "field-row-item";
      row.innerHTML = `
        <input type="text" class="repeater-input row-degree" placeholder="Degree / Certificate">
        <input type="text" class="repeater-input row-inst" placeholder="School / Board / University">
        <input type="text" class="repeater-input row-year" placeholder="Year">
        <input type="text" class="repeater-input row-score" placeholder="Score / %">
        <button type="button" class="btn-remove-row" onclick="removeRow(this)">&times;</button>
      `;
      container.appendChild(row);
    }

    function removeRow(btn) {
      if (container.children.length <= 1) {
        alert("You must keep at least one qualification row!");
        return;
      }
      btn.parentElement.remove();
    }

    function clearAllRows() {
      container.innerHTML = "";
      addEducationRow();
    }

    function serializeFormData() {
      const rows = document.querySelectorAll(".field-row-item");
      const list = [];

      rows.forEach(r => {
        list.push({
          degree: r.querySelector(".row-degree").value.trim(),
          institution: r.querySelector(".row-inst").value.trim(),
          year: r.querySelector(".row-year").value.trim(),
          score: r.querySelector(".row-score").value.trim()
        });
      });

      const payload = {
        student: "Shaik Roshan",
        rollNumber: "[TO BE FILLED BY STUDENT]",
        qualifications: list,
        submittedAt: new Date().toISOString()
      };

      document.getElementById("payloadOutput").textContent = JSON.stringify(payload, null, 2);
    }
