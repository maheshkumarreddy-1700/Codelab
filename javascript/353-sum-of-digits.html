Sum of digits

Question: Write a JavaScript program to sum of digits.
▶ Live Output
Working example
JavaScript Code
function calcSumDigits() {
      const input = document.getElementById('digits-num').value.trim();
      const res = document.getElementById('res-digits');
      let n = Math.abs(parseInt(input));
      if (isNaN(n)) {
        res.textContent = "Please enter an integer.";
        return;
      }
      if (n === 0) {
        res.textContent = "Sum of digits of 0 = 0";
        return;
      }
      let digits = [];
      let temp = n;
      let sum = 0;
      while (temp > 0) {
        let d = temp % 10;
        digits.unshift(d);
        sum += d;
        temp = Math.floor(temp / 10);
      }
      res.innerHTML = `${digits.join(' + ')} = <strong>${sum}</strong>`;
    }
