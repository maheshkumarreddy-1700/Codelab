Instant validation

Question: Write a JavaScript program to instant validation.

function updateField(rowId, iconId, hintId, ok, goodText, badText) {
      const row = document.getElementById(rowId);
      const icon = document.getElementById(iconId);
      const hint = document.getElementById(hintId);

      if (ok) {
        row.className = "field-row valid";
        icon.textContent = "✔";
        icon.style.color = "#10b981";
        hint.textContent = goodText;
      } else {
        row.className = "field-row invalid";
        icon.textContent = "✖";
        icon.style.color = "#ef4444";
        hint.textContent = badText;
      }
    }

    function validateUser(input) {
      const val = input.value.trim();
      const ok = /^[a-zA-Z0-9_]{4,16}$/.test(val);
      updateField("rowUsername", "iconUsername", "hintUsername", ok, "Username is available!", "Must be 4-16 alphanumeric characters or _");
    }

    function validateEmailField(input) {
      const val = input.value.trim();
      const ok = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(val);
      updateField("rowEmail", "iconEmail", "hintEmail", ok, "Email syntax verified!", "Invalid email format (e.g. name@domain.com)");
    }

    function validatePinField(input) {
      const val = input.value.trim();
      const ok = /^\d{6}$/.test(val);
      updateField("rowPin", "iconPin", "hintPin", ok, "6-digit PIN accepted!", "Must be exactly 6 numeric digits");
    }
