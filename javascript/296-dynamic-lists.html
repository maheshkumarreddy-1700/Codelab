Dynamic lists

Question: Write a JavaScript program to dynamic lists.
▶ Live Output
Working example
JavaScript Code
const list = document.getElementById("taskList");
    const input = document.getElementById("newItemInput");
    const countDisplay = document.getElementById("countDisplay");

    function updateCount() {
      countDisplay.textContent = list.children.length;
    }

    function addItem() {
      const text = input.value.trim();
      if (!text) {
        alert("Please enter task description!");
        return;
      }

      const li = document.createElement("li");
      li.className = "task-item";
      li.innerHTML = `
        <span>${text}</span>
        <div class="task-actions">
          <button class="btn-icon" title="Toggle Complete" onclick="toggleTask(this)">✔️</button>
          <button class="btn-icon" title="Delete" onclick="deleteTask(this)">🗑️</button>
        </div>
      `;
      list.appendChild(li);
      input.value = "";
      updateCount();
    }

    function toggleTask(btn) {
      const item = btn.closest(".task-item");
      item.classList.toggle("completed");
    }

    function deleteTask(btn) {
      const item = btn.closest(".task-item");
      item.remove();
      updateCount();
    }

    function clearAll() {
      const completed = list.querySelectorAll(".task-item.completed");
      completed.forEach(item => item.remove());
      updateCount();
    }

    updateCount();
