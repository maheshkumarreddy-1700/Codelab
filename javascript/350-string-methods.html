function inspectString() {
      const s = document.getElementById('str-input').value;
      const t = s.trim();

      const data = [
        { call: ".length", desc: "Total character count (including whitespace)", val: s.length },
        { call: ".trim()", desc: "Strips leading and trailing whitespace", val: `"${t}"` },
        { call: ".toUpperCase()", desc: "Converts entire string to uppercase", val: `"${s.toUpperCase()}"` },
        { call: ".toLowerCase()", desc: "Converts entire string to lowercase", val: `"${s.toLowerCase()}"` },
        { call: ".charAt(0)", desc: "First non-empty character", val: `'${t.charAt(0)}'` },
        { call: ".slice(0, 15)", desc: "Extracts characters between indices 0 and 15", val: `"${t.slice(0, 15)}"` },
        { call: ".includes('Practical')", desc: "Checks substring presence", val: t.includes("Practical") },
        { call: ".replace('Record', 'Portfolio')", desc: "Substitutes matching pattern", val: `"${t.replace("Record", "Portfolio")}"` },
        { call: ".split(' ')", desc: "Splits words into an Array", val: JSON.stringify(t.split(/\s+/)) }
      ];

      document.getElementById('string-table-body').innerHTML = data.map(row => `
        <tr>
          <td style="padding: 0.6rem; border: 1px solid var(--border); font-family: var(--font-mono); font-weight: 600; color: #ca8a04;">${row.call}</td>
          <td style="padding: 0.6rem; border: 1px solid var(--border); color: #64748b;">${row.desc}</td>
          <td style="padding: 0.6rem; border: 1px solid var(--border); font-family: var(--font-mono); font-weight: 700; color: #0f172a;">${row.val}</td>
        </tr>
      `).join('');
    }
    inspectString();
