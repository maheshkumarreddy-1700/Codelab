Drag and drop

Question: Write a JavaScript program to drag and drop.
▶ Live Output
Working example
JavaScript Code
function updateCounts() {
      document.getElementById("countTodo").textContent = document.getElementById("zoneTodo").children.length;
      document.getElementById("countProgress").textContent = document.getElementById("zoneProgress").children.length;
      document.getElementById("countDone").textContent = document.getElementById("zoneDone").children.length;
    }

    const cards = document.querySelectorAll(".task-card");
    cards.forEach(card => {
      card.addEventListener("dragstart", (e) => {
        e.dataTransfer.setData("text/plain", card.id);
        card.classList.add("dragging");
      });

      card.addEventListener("dragend", () => {
        card.classList.remove("dragging");
        updateCounts();
      });
    });

    function handleDragOver(e) {
      e.preventDefault();
      e.currentTarget.classList.add("drag-over");
    }

    function handleDragLeave(e) {
      e.currentTarget.classList.remove("drag-over");
    }

    function handleDrop(e) {
      e.preventDefault();
      const zone = e.currentTarget;
      zone.classList.remove("drag-over");

      const id = e.dataTransfer.getData("text/plain");
      const card = document.getElementById(id);
      if (card) {
        zone.appendChild(card);
        updateCounts();
      }
    }

    updateCounts();
