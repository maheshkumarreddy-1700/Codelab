Dynamic dropdown

Question: Write a JavaScript program to dynamic dropdown.
▶ Live Output
Working example
JavaScript Code
const select = document.getElementById("courseSelect");
    const selectedName = document.getElementById("selectedName");
    const selectedVal = document.getElementById("selectedVal");

    function handleCourseChange() {
      if (select.selectedIndex <= 0) {
        selectedName.textContent = "None Selected";
        selectedVal.textContent = "Value: -";
        return;
      }
      const opt = select.options[select.selectedIndex];
      selectedName.textContent = opt.text;
      selectedVal.textContent = `Value: "${opt.value}" (Index: ${select.selectedIndex})`;
    }

    function addCourseOption() {
      const text = document.getElementById("newCourseText").value.trim();
      const val = document.getElementById("newCourseVal").value.trim();

      if (!text || !val) {
        alert("Please provide both course title and code!");
        return;
      }

      const opt = new Option(text, val);
      select.add(opt);
      select.value = val;
      handleCourseChange();

      document.getElementById("newCourseText").value = "";
      document.getElementById("newCourseVal").value = "";
    }

    function removeSelectedOption() {
      if (select.selectedIndex <= 0) {
        alert("Select an elective to remove!");
        return;
      }
      select.remove(select.selectedIndex);
      handleCourseChange();
    }

    function resetOptions() {
      select.innerHTML = `
        <option value="">-- Choose an Elective Course --</option>
        <option value="cs-401">Full Stack Web Development</option>
        <option value="cs-402">Cloud Computing & DevOps</option>
        <option value="cs-403">Machine Learning Fundamentals</option>
      `;
      handleCourseChange();
    }
