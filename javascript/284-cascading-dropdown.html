Cascading dropdown

Question: Write a JavaScript program to cascading dropdown.
▶ Live Output
Working example
JavaScript Code
const geoData = {
      "India": {
        "Telangana": ["Hyderabad", "Warangal", "Karimnagar", "Nizamabad"],
        "Karnataka": ["Bengaluru", "Mysuru", "Mangaluru", "Hubballi"],
        "Maharashtra": ["Mumbai", "Pune", "Nagpur", "Nashik"],
        "Tamil Nadu": ["Chennai", "Coimbatore", "Madurai"]
      },
      "United States": {
        "California": ["Los Angeles", "San Francisco", "San Jose", "San Diego"],
        "Texas": ["Houston", "Austin", "Dallas", "San Antonio"],
        "New York": ["New York City", "Buffalo", "Rochester"]
      },
      "United Kingdom": {
        "England": ["London", "Manchester", "Birmingham", "Liverpool"],
        "Scotland": ["Edinburgh", "Glasgow", "Aberdeen"]
      }
    };

    const countrySel = document.getElementById("countrySelect");
    const stateSel = document.getElementById("stateSelect");
    const citySel = document.getElementById("citySelect");
    const summary = document.getElementById("selectionSummary");

    function initCountries() {
      Object.keys(geoData).forEach(country => {
        countrySel.add(new Option(country, country));
      });
    }

    function onCountrySelected() {
      const country = countrySel.value;
      stateSel.innerHTML = '<option value="">-- Choose State --</option>';
      citySel.innerHTML = '<option value="">-- Choose City --</option>';
      citySel.disabled = true;

      if (country && geoData[country]) {
        stateSel.disabled = false;
        Object.keys(geoData[country]).forEach(st => {
          stateSel.add(new Option(st, st));
        });
        summary.innerHTML = `Selected Country: <strong>${country}</strong>. Now choose a state.`;
      } else {
        stateSel.disabled = true;
        summary.innerHTML = "Please select a country to begin.";
      }
    }

    function onStateSelected() {
      const country = countrySel.value;
      const state = stateSel.value;
      citySel.innerHTML = '<option value="">-- Choose City --</option>';

      if (state && geoData[country][state]) {
        citySel.disabled = false;
        geoData[country][state].forEach(c => {
          citySel.add(new Option(c, c));
        });
        summary.innerHTML = `Country: <strong>${country}</strong> &rsaquo; State: <strong>${state}</strong>. Now choose a city.`;
      } else {
        citySel.disabled = true;
      }
    }

    function onCitySelected() {
      const country = countrySel.value;
      const state = stateSel.value;
      const city = citySel.value;

      if (city) {
        summary.innerHTML = `Selected Location: <strong style="color:#38bdf8;">${city}, ${state}, ${country}</strong>`;
      }
    }

    initCountries();
