Reaction time game

Question: Write a JavaScript program to reaction time game.
▶ Live Output
Working example
JavaScript Code
let state = "idle";
    let startTime = 0;
    let timeoutId = null;
    let bestTime = Infinity;

    const box = document.getElementById("targetBox");
    const boxIcon = document.getElementById("boxIcon");
    const boxTitle = document.getElementById("boxTitle");
    const boxSub = document.getElementById("boxSubtitle");
    const resultDisplay = document.getElementById("reactionResult");
    const bestDisplay = document.getElementById("bestScore");

    function setBoxState(newState, title, sub, icon = "⚡") {
      state = newState;
      box.className = `reaction-target-box state-${newState}`;
      boxIcon.textContent = icon;
      boxTitle.textContent = title;
      boxSub.textContent = sub;
      resultDisplay.style.display = "none";
    }

    function handleBoxClick() {
      if (state === "idle" || state === "result") {
        setBoxState("waiting", "Wait for Green...", "Do NOT click yet!", "🛑");
        const delay = Math.floor(Math.random() * 3000 + 1500);

        timeoutId = setTimeout(() => {
          state = "ready";
          startTime = performance.now();
          box.className = "reaction-target-box state-ready";
          boxIcon.textContent = "🟢";
          boxTitle.textContent = "CLICK NOW!";
          boxSub.textContent = "Click as fast as possible!";
        }, delay);

      } else if (state === "waiting") {
        clearTimeout(timeoutId);
        setBoxState("result", "Too Early! ⚠️", "You clicked before the screen turned green. Click to retry.", "❌");

      } else if (state === "ready") {
        const reactionMs = Math.round(performance.now() - startTime);
        if (reactionMs < bestTime) {
          bestTime = reactionMs;
          bestDisplay.textContent = `${bestTime} ms`;
        }

        setBoxState("result", "Great Reflexes!", "Click anywhere to test again.", "🏆");
        resultDisplay.textContent = `${reactionMs} ms`;
        resultDisplay.style.display = "block";
      }
    }
