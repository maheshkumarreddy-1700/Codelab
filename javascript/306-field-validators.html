Field validators

Question: Write a JavaScript program to field validators.
▶ Live Output
Working example
JavaScript Code
function updateFieldStatus(stEl, isValid, label) {
      if (isValid) {
        stEl.textContent = `✔ Valid ${label}`;
        stEl.className = "feedback-note status-ok";
      } else {
        stEl.textContent = `✖ Invalid`;
        stEl.className = "feedback-note status-bad";
      }
    }

    function testName() {
      const val = document.getElementById("inpName").value.trim();
      const ok = /^[a-zA-Z\s]{3,}$/.test(val);
      updateFieldStatus(document.getElementById("stName"), ok, "Name");
    }

    function testPhone() {
      const val = document.getElementById("inpPhone").value.trim();
      const ok = /^[6-9]\d{9}$/.test(val);
      updateFieldStatus(document.getElementById("stPhone"), ok, "Phone");
    }

    function testAge() {
      const val = parseInt(document.getElementById("inpAge").value);
      const ok = !isNaN(val) && val >= 18 && val <= 60;
      updateFieldStatus(document.getElementById("stAge"), ok, "Age");
    }

    function testPin() {
      const val = document.getElementById("inpPin").value.trim();
      const ok = /^[1-9][0-9]{5}$/.test(val);
      updateFieldStatus(document.getElementById("stPin"), ok, "PIN");
    }
