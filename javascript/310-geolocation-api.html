Geolocation api

Question: Write a JavaScript program to geolocation api.

function requestLocation() {
      const statusBadge = document.getElementById("geoStatus");
      const btn = document.getElementById("btnGetLoc");

      if (!navigator.geolocation) {
        statusBadge.textContent = "Unsupported";
        statusBadge.className = "badge badge-warning";
        alert("Geolocation API is not supported in this browser environment.");
        return;
      }

      statusBadge.textContent = "Requesting GPS...";
      statusBadge.className = "badge badge-warning";
      btn.disabled = true;

      navigator.geolocation.getCurrentPosition(
        (pos) => {
          const lat = pos.coords.latitude.toFixed(6);
          const lng = pos.coords.longitude.toFixed(6);
          const acc = Math.round(pos.coords.accuracy);

          document.getElementById("valLat").textContent = `${lat}°`;
          document.getElementById("valLng").textContent = `${lng}°`;
          document.getElementById("valAcc").textContent = `± ${acc} meters`;

          const mapLink = document.getElementById("openMapLink");
          mapLink.href = `https://www.openstreetmap.org/?mlat=${lat}&mlon=${lng}#map=15/${lat}/${lng}`;
          document.getElementById("mapBtnWrap").style.display = "block";

          statusBadge.textContent = "Coordinates Acquired";
          statusBadge.className = "badge badge-success";
          btn.disabled = false;
        },
        (err) => {
          statusBadge.textContent = "Permission Denied / Error";
          statusBadge.className = "badge badge-secondary";
          btn.disabled = false;

          // Provide fallback coordinates (e.g. Hyderabad Tech City)
          document.getElementById("valLat").textContent = "17.3850°";
          document.getElementById("valLng").textContent = "78.4867°";
          document.getElementById("valAcc").textContent = "Default Lab Coordinates (Simulated)";
          const mapLink = document.getElementById("openMapLink");
          mapLink.href = `https://www.openstreetmap.org/?mlat=17.3850&mlon=78.4867#map=14/17.3850/78.4867`;
          document.getElementById("mapBtnWrap").style.display = "block";
        },
        { enableHighAccuracy: true, timeout: 8000 }
      );
    }
