Stopwatch

Question: Write a JavaScript program to stopwatch.
▶ Live Output
Working example
JavaScript Code
let startTime = 0;
    let elapsedTime = 0;
    let timer = null;
    let lapCount = 0;

    const mEl = document.getElementById("swMinutes");
    const sEl = document.getElementById("swSeconds");
    const msEl = document.getElementById("swMillis");
    const lapsBox = document.getElementById("lapsBox");

    const btnStart = document.getElementById("btnSwStart");
    const btnStop = document.getElementById("btnSwStop");
    const btnLap = document.getElementById("btnSwLap");
    const badge = document.getElementById("swBadge");

    function renderTime(ms) {
      const minutes = Math.floor(ms / (1000 * 60));
      const seconds = Math.floor((ms % (1000 * 60)) / 1000);
      const hundredths = Math.floor((ms % 1000) / 10);

      mEl.textContent = String(minutes).padStart(2, '0');
      sEl.textContent = String(seconds).padStart(2, '0');
      msEl.textContent = String(hundredths).padStart(2, '0');

      return `${mEl.textContent}:${sEl.textContent}.${msEl.textContent}`;
    }

    function startSw() {
      if (timer) return;
      startTime = Date.now() - elapsedTime;
      timer = setInterval(() => {
        elapsedTime = Date.now() - startTime;
        renderTime(elapsedTime);
      }, 10);

      btnStart.disabled = true;
      btnStop.disabled = false;
      btnLap.disabled = false;
      badge.textContent = "Running";
      badge.className = "badge badge-primary";
    }

    function stopSw() {
      clearInterval(timer);
      timer = null;
      btnStart.disabled = false;
      btnStop.disabled = true;
      btnLap.disabled = true;
      badge.textContent = "Stopped";
      badge.className = "badge badge-warning";
    }

    function resetSw() {
      stopSw();
      elapsedTime = 0;
      renderTime(0);
      lapCount = 0;
      lapsBox.innerHTML = '<div style="padding: 1rem; color: #64748b; font-size: 0.85rem;">No lap recorded yet. Click \'Lap\' while running.</div>';
      badge.textContent = "Ready";
      badge.className = "badge badge-success";
    }

    function recordLap() {
      if (!timer) return;
      lapCount++;
      if (lapCount === 1) lapsBox.innerHTML = "";

      const row = document.createElement("div");
      row.className = "lap-row";
      row.innerHTML = `<span>Lap ${lapCount}</span><span>${renderTime(elapsedTime)}</span>`;
      lapsBox.prepend(row);
    }
