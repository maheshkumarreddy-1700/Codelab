Factorial

Question: Write a JavaScript program to factorial.

function computeFactorial() {
      const n = parseInt(document.getElementById('fact-num').value);
      const res = document.getElementById('res-fact');
      if (isNaN(n) || n < 0) {
        res.textContent = "Please enter a valid non-negative integer (n >= 0).";
        return;
      }
      if (n > 20) {
        // Use BigInt for very large factorials
        let fact = 1n;
        for (let i = 2n; i <= BigInt(n); i++) fact *= i;
        res.innerHTML = `${n}! = <strong>${fact.toString()}</strong> <br><small style="color: #64748b;">(Calculated via BigInt precision)</small>`;
        return;
      }
      if (n === 0) {
        res.innerHTML = `0! = <strong>1</strong> (By mathematical definition)`;
        return;
      }
      let fact = 1;
      let parts = [];
      for (let i = n; i >= 1; i--) {
        fact *= i;
        parts.push(i);
      }
      res.innerHTML = `${n}! = ${parts.join(' &times; ')} = <strong>${fact.toLocaleString()}</strong>`;
    }
