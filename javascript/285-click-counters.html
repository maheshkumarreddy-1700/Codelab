Click counters

Question: Write a JavaScript program to click counters.
▶ Live Output
Working example
JavaScript Code
let clicks = 0;
    let timeLeft = 5.0;
    let timer = null;
    let isRunning = false;

    const clickArea = document.getElementById("clickArea");
    const clickPrompt = document.getElementById("clickPrompt");
    const timeEl = document.getElementById("timeRemaining");
    const clicksEl = document.getElementById("totalClicks");
    const cpsEl = document.getElementById("cpsScore");

    function handleClick() {
      if (!isRunning && timeLeft === 5.0) {
        isRunning = true;
        clickPrompt.textContent = "Click as fast as you can!";
        startTimer();
      }

      if (isRunning) {
        clicks++;
        clicksEl.textContent = clicks;
        const elapsed = 5.0 - timeLeft;
        if (elapsed > 0) {
          cpsEl.textContent = (clicks / elapsed).toFixed(1);
        }
      }
    }

    function startTimer() {
      timer = setInterval(() => {
        timeLeft -= 0.1;
        if (timeLeft <= 0) {
          timeLeft = 0;
          clearInterval(timer);
          isRunning = false;
          timeEl.textContent = "0.0s";
          clickPrompt.textContent = `Completed! ${clicks} Clicks`;
          clickArea.disabled = true;
          cpsEl.textContent = (clicks / 5.0).toFixed(2);
          return;
        }
        timeEl.textContent = `${timeLeft.toFixed(1)}s`;
      }, 100);
    }

    function resetTest() {
      clearInterval(timer);
      timer = null;
      isRunning = false;
      clicks = 0;
      timeLeft = 5.0;
      clickArea.disabled = false;
      clickPrompt.textContent = "Click here to start test!";
      timeEl.textContent = "5.0s";
      clicksEl.textContent = "0";
      cpsEl.textContent = "0.0";
    }
