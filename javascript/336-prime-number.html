Prime number

Question: Write a JavaScript program to prime number.

function checkPrime() {
      const n = parseInt(document.getElementById('prime-input').value);
      const res = document.getElementById('res-prime');
      if (isNaN(n)) {
        res.textContent = "Please enter an integer.";
        return;
      }
      if (n <= 1) {
        res.style.color = "#ef4444";
        res.textContent = `${n} is NOT a prime number (Primes must be integers > 1).`;
        return;
      }
      let divisor = 0;
      const limit = Math.floor(Math.sqrt(n));
      for (let i = 2; i <= limit; i++) {
        if (n % i === 0) {
          divisor = i;
          break;
        }
      }
      if (divisor > 0) {
        res.style.color = "#ef4444";
        res.innerHTML = `${n} is a <strong>COMPOSITE</strong> number (Divisible by ${divisor}: ${divisor} &times; ${n / divisor} = ${n}).`;
      } else {
        res.style.color = "#16a34a";
        res.innerHTML = `${n} is a <strong>PRIME</strong> number! (No integer divisors found up to &radic;${n} &approx; ${limit}).`;
      }
    }
