Theme switcher storage

Question: Write a JavaScript program to theme switcher storage.
▶ Live Output
Working example
JavaScript Code
const playground = document.getElementById("playground");
    const themeHeader = document.getElementById("themeHeader");
    const badge = document.getElementById("activeThemeBadge");

    function applyTheme(theme) {
      playground.className = `theme-playground-box theme-${theme}`;
      localStorage.setItem("lab_theme_pref", theme);

      document.querySelectorAll(".theme-pill-btn").forEach(btn => btn.classList.remove("active"));
      if (theme === "dark") {
        document.getElementById("btnThemeDark").classList.add("active");
        themeHeader.textContent = "🌙 Dark Mode Active";
        badge.textContent = "Active: Dark";
      } else if (theme === "light") {
        document.getElementById("btnThemeLight").classList.add("active");
        themeHeader.textContent = "☀️ Light Mode Active";
        badge.textContent = "Active: Light";
      } else if (theme === "cyber") {
        document.getElementById("btnThemeCyber").classList.add("active");
        themeHeader.textContent = "👾 Cyberpunk Neon Active";
        badge.textContent = "Active: Cyberpunk";
      }
    }

    // Load initial saved theme
    const saved = localStorage.getItem("lab_theme_pref") || "dark";
    applyTheme(saved);
