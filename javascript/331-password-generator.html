Password generator

Question: Write a JavaScript program to password generator.
▶ Live Output
Working example
JavaScript Code
(function() {
      const chars = {
        upper: 'ABCDEFGHIJKLMNOPQRSTUVWXYZ',
        lower: 'abcdefghijklmnopqrstuvwxyz',
        numbers: '0123456789',
        symbols: '!@#$%^&*()_+~`|}{[]:;?><,./-='
      };

      const passOutput = document.getElementById('passOutput');
      const btnCopyPass = document.getElementById('btnCopyPass');
      const passLength = document.getElementById('passLength');
      const lengthDisplay = document.getElementById('lengthDisplay');
      const chkUpper = document.getElementById('chkUpper');
      const chkLower = document.getElementById('chkLower');
      const chkNumbers = document.getElementById('chkNumbers');
      const chkSymbols = document.getElementById('chkSymbols');
      const btnGenPass = document.getElementById('btnGenPass');
      const strengthBar = document.getElementById('strengthBar');
      const strengthText = document.getElementById('strengthText');
      const toastNotification = document.getElementById('toastNotification');

      passLength.addEventListener('input', (e) => {
        lengthDisplay.textContent = e.target.value;
        generatePassword();
      });

      [chkUpper, chkLower, chkNumbers, chkSymbols].forEach(cb => {
        cb.addEventListener('change', generatePassword);
      });

      function generatePassword() {
        const length = parseInt(passLength.value);
        let pool = '';
        let guaranteed = [];

        if (chkUpper.checked) {
          pool += chars.upper;
          guaranteed.push(chars.upper[Math.floor(Math.random() * chars.upper.length)]);
        }
        if (chkLower.checked) {
          pool += chars.lower;
          guaranteed.push(chars.lower[Math.floor(Math.random() * chars.lower.length)]);
        }
        if (chkNumbers.checked) {
          pool += chars.numbers;
          guaranteed.push(chars.numbers[Math.floor(Math.random() * chars.numbers.length)]);
        }
        if (chkSymbols.checked) {
          pool += chars.symbols;
          guaranteed.push(chars.symbols[Math.floor(Math.random() * chars.symbols.length)]);
        }

        if (!pool) {
          passOutput.textContent = 'Select at least 1 option';
          strengthBar.style.width = '0%';
          strengthText.textContent = 'None';
          strengthText.style.color = 'var(--text-secondary)';
          return;
        }

        let result = [...guaranteed];
        for (let i = guaranteed.length; i < length; i++) {
          result.push(pool[Math.floor(Math.random() * pool.length)]);
        }

        result.sort(() => Math.random() - 0.5);
        const password = result.join('');
        passOutput.textContent = password;
        evaluateStrength(password, pool.length);
      }

      function evaluateStrength(pass, poolSize) {
        let score = 0;
        if (pass.length >= 8) score++;
        if (pass.length >= 14) score++;
        if (poolSize >= 36) score++;
        if (poolSize >= 62) score++;

        if (score <= 1) {
          strengthBar.style.width = '25%';
          strengthBar.style.background = '#ef4444';
          strengthText.textContent = 'Weak';
          strengthText.style.color = '#ef4444';
        } else if (score === 2) {
          strengthBar.style.width = '50%';
          strengthBar.style.background = '#f59e0b';
          strengthText.textContent = 'Medium';
          strengthText.style.color = '#f59e0b';
        } else if (score === 3) {
          strengthBar.style.width = '75%';
          strengthBar.style.background = '#3b82f6';
          strengthText.textContent = 'Strong';
          strengthText.style.color = '#3b82f6';
        } else {
          strengthBar.style.width = '100%';
          strengthBar.style.background = '#10b981';
          strengthText.textContent = 'Very Strong';
          strengthText.style.color = '#10b981';
        }
      }

      btnGenPass.addEventListener('click', generatePassword);

      btnCopyPass.addEventListener('click', () => {
        const text = passOutput.textContent;
        if (!text || text.includes('Select')) return;
        navigator.clipboard.writeText(text).then(() => {
          toastNotification.textContent = 'Password copied to clipboard!';
          toastNotification.classList.add('show');
          setTimeout(() => toastNotification.classList.remove('show'), 2000);
        });
      });

      // Initial run
      generatePassword();
    })();
