function triggerAlert() {
      alert("Hello Shaik Roshan! This is a standard modal window.alert() box.");
      document.getElementById("outAlert").textContent = "Return: undefined (dismissed)";
    }

    function triggerConfirm() {
      const res = confirm("Do you want to submit the Web Technology Laboratory Record now?");
      document.getElementById("outConfirm").textContent = `Return: ${res} (${typeof res})`;
    }

    function triggerPrompt() {
      const res = prompt("Enter student roll number:", "22CS042");
      if (res === null) {
        document.getElementById("outPrompt").textContent = `Return: null (User clicked Cancel)`;
      } else {
        document.getElementById("outPrompt").textContent = `Return: "${res}"`;
      }
    }
