Location history

Question: Write a JavaScript program to location history.
▶ Live Output
Working example
JavaScript Code
function renderLocation() {
      document.getElementById("locHref").textContent = window.location.href;
      document.getElementById("locProtocol").textContent = window.location.protocol;
      document.getElementById("locHost").textContent = window.location.host || "(Local file system)";
      document.getElementById("locPath").textContent = window.location.pathname;
      document.getElementById("locHash").textContent = window.location.hash || "(empty)";
      document.getElementById("histLength").textContent = window.history.length;
    }

    function setSampleHash() {
      window.location.hash = "student-shaik-roshan";
      renderLocation();
    }

    function simulatePushState() {
      window.history.pushState({ id: 42 }, "Lab Demo", "?query=webtech-lab-2026");
      renderLocation();
      document.getElementById("pushNotice").style.display = "block";
    }

    renderLocation();
