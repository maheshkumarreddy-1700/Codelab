Tic tac toe

Question: Write a JavaScript program to tic tac toe.
▶ Live Output
Working example
JavaScript Code
let board = ["", "", "", "", "", "", "", "", ""];
    let currentPlayer = "X";
    let isGameOver = false;
    let score = { X: 0, O: 0, ties: 0 };

    const winPatterns = [
      [0,1,2], [3,4,5], [6,7,8],
      [0,3,6], [1,4,7], [2,5,8],
      [0,4,8], [2,4,6]
    ];

    const statusBanner = document.getElementById("gameStatus");
    const aiCheck = document.getElementById("aiModeCheck");

    function makeMove(idx) {
      if (board[idx] !== "" || isGameOver) return;

      board[idx] = currentPlayer;
      const btn = document.querySelector(`[data-idx="${idx}"]`);
      btn.textContent = currentPlayer;
      btn.classList.add(currentPlayer === "X" ? "x-player" : "o-player");
      btn.disabled = true;

      const winResult = checkWinner();
      if (winResult) {
        handleGameEnd(winResult);
        return;
      }

      currentPlayer = currentPlayer === "X" ? "O" : "X";
      statusBanner.textContent = `Player ${currentPlayer}'s Turn`;

      if (aiCheck.checked && currentPlayer === "O" && !isGameOver) {
        setTimeout(makeAiMove, 300);
      }
    }

    function makeAiMove() {
      // Find available empty indices
      const available = [];
      board.forEach((val, i) => { if (val === "") available.push(i); });
      if (available.length === 0) return;

      const choice = available[Math.floor(Math.random() * available.length)];
      makeMove(choice);
    }

    function checkWinner() {
      for (const [a, b, c] of winPatterns) {
        if (board[a] && board[a] === board[b] && board[a] === board[c]) {
          return { winner: board[a], pattern: [a, b, c] };
        }
      }
      if (board.every(cell => cell !== "")) {
        return { winner: "tie" };
      }
      return null;
    }

    function handleGameEnd(res) {
      isGameOver = true;
      if (res.winner === "tie") {
        statusBanner.textContent = "🤝 It's a Drawn Tie!";
        score.ties++;
        document.getElementById("scoreTies").textContent = score.ties;
      } else {
        statusBanner.textContent = `🎉 Player ${res.winner} Wins!`;
        score[res.winner]++;
        document.getElementById(`score${res.winner}`).textContent = score[res.winner];
        res.pattern.forEach(i => {
          document.querySelector(`[data-idx="${i}"]`).classList.add("winner-cell");
        });
      }
    }

    function resetGame() {
      board = ["", "", "", "", "", "", "", "", ""];
      currentPlayer = "X";
      isGameOver = false;
      statusBanner.textContent = "Player X's Turn";

      document.querySelectorAll(".cell-btn").forEach(btn => {
        btn.textContent = "";
        btn.disabled = false;
        btn.className = "cell-btn";
      });
    }
