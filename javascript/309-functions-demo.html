function calculateSquare(n) { return n * n; }
    const calculateCube = function(n) { return n * n * n; };
    const calculateDouble = n => n * 2;

    function runFunctions() {
      const x = parseFloat(document.getElementById('base-val').value) || 0;
      const res = document.getElementById('res-functions');
      res.innerHTML = `
        1. Function Declaration: calculateSquare(${x}) = <strong>${calculateSquare(x)}</strong><br>
        2. Function Expression: calculateCube(${x}) = <strong>${calculateCube(x)}</strong><br>
        3. ES6 Arrow Function: calculateDouble(${x}) = <strong>${calculateDouble(x)}</strong>
      `;
    }
    runFunctions();
