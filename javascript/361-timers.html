// setTimeout Demo
    let timeoutId = null;
    let animTimer = null;
    const progressEl = document.getElementById("timeoutProgress");
    const timeoutStatus = document.getElementById("timeoutStatus");

    function startTimeoutDemo() {
      cancelTimeoutDemo();
      timeoutStatus.textContent = "Running (3.0s)...";
      timeoutStatus.style.color = "#f59e0b";
      progressEl.style.width = "0%";

      let elapsed = 0;
      animTimer = setInterval(() => {
        elapsed += 50;
        const pct = Math.min(100, (elapsed / 3000) * 100);
        progressEl.style.width = `${pct}%`;
      }, 50);

      timeoutId = setTimeout(() => {
        clearInterval(animTimer);
        progressEl.style.width = "100%";
        timeoutStatus.textContent = "🎉 Timer Fired!";
        timeoutStatus.style.color = "#10b981";
      }, 3000);
    }

    function cancelTimeoutDemo() {
      clearTimeout(timeoutId);
      clearInterval(animTimer);
      progressEl.style.width = "0%";
      timeoutStatus.textContent = "Cancelled / Idle";
      timeoutStatus.style.color = "#38bdf8";
    }

    // setInterval Demo
    let intervalId = null;
    let tick = 0;
    const intervalDisplay = document.getElementById("intervalDisplay");

    function startIntervalDemo() {
      if (intervalId) return;
      intervalId = setInterval(() => {
        tick++;
        intervalDisplay.textContent = `${tick}s`;
      }, 1000);
    }

    function stopIntervalDemo() {
      clearInterval(intervalId);
      intervalId = null;
    }

    function resetIntervalDemo() {
      stopIntervalDemo();
      tick = 0;
      intervalDisplay.textContent = "0s";
    }
