Even odd

Question: Write a JavaScript program to even odd.

function checkParity() {
      const n = parseInt(document.getElementById('parity-num').value);
      if (isNaN(n)) {
        document.getElementById('res-parity').textContent = "Please enter an integer.";
        return;
      }
      const isEven = (n % 2 === 0);
      const resEl = document.getElementById('res-parity');
      if (isEven) {
        resEl.style.color = "#16a34a";
        resEl.textContent = `${n} is an EVEN number (${n} % 2 === 0).`;
      } else {
        resEl.style.color = "#ea580c";
        resEl.textContent = `${n} is an ODD number (${n} % 2 !== 0).`;
      }
    }
