Live counters

Question: Write a JavaScript program to live counters.
▶ Live Output
Working example
JavaScript Code
const textarea = document.getElementById("liveTextArea");
    const MAX_LIMIT = 250;

    function updateMetrics() {
      const text = textarea.value;
      const chars = text.length;
      const words = text.trim() === "" ? 0 : text.trim().split(/\s+/).length;
      const sentences = text.trim() === "" ? 0 : (text.match(/[\w|\)][.?!]+(\s|$)/g) || []).length;
      const readSeconds = Math.ceil((words / 200) * 60);

      document.getElementById("countChars").textContent = chars;
      document.getElementById("countWords").textContent = words;
      document.getElementById("countSentences").textContent = sentences;
      document.getElementById("countReadingTime").textContent = readSeconds < 60 ? `${readSeconds}s` : `${Math.ceil(readSeconds/60)}m`;

      // Progress bar & limit indicator
      const percent = Math.min(100, Math.round((chars / MAX_LIMIT) * 100));
      const fill = document.getElementById("progressFill");
      fill.style.width = `${percent}%`;

      const limitLabel = document.getElementById("charLimitIndicator");
      limitLabel.textContent = `${chars} / ${MAX_LIMIT} characters`;

      if (chars > MAX_LIMIT) {
        textarea.classList.add("limit-warning");
        fill.style.backgroundColor = "#ef4444";
        limitLabel.style.color = "#ef4444";
      } else {
        textarea.classList.remove("limit-warning");
        fill.style.backgroundColor = "#38bdf8";
        limitLabel.style.color = "#94a3b8";
      }
    }

    function clearText() {
      textarea.value = "";
      updateMetrics();
    }
