Currency converter

Question: Write a JavaScript program to currency converter.
▶ Live Output
Working example
JavaScript Code
(function() {
      const exchangeRates = {
        USD: 1.0,
        EUR: 0.92,
        GBP: 0.79,
        INR: 83.20,
        JPY: 155.40,
        CAD: 1.36,
        AUD: 1.52
      };

      const inpAmount = document.getElementById('inpAmount');
      const selFrom = document.getElementById('selFrom');
      const selTo = document.getElementById('selTo');
      const btnSwap = document.getElementById('btnSwap');
      const dispFromSummary = document.getElementById('dispFromSummary');
      const dispToResult = document.getElementById('dispToResult');
      const dispRateNote = document.getElementById('dispRateNote');

      function convert() {
        const amount = parseFloat(inpAmount.value) || 0;
        const from = selFrom.value;
        const to = selTo.value;

        const rateInUSD = 1 / exchangeRates[from];
        const converted = amount * rateInUSD * exchangeRates[to];
        const unitRate = (1 * rateInUSD * exchangeRates[to]);

        dispFromSummary.textContent = `${amount.toLocaleString()} ${from} =`;
        dispToResult.textContent = `${converted.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} ${to}`;
        dispRateNote.textContent = `1 ${from} = ${unitRate.toFixed(4)} ${to} • Mid-market exchange rate`;
      }

      btnSwap.addEventListener('click', () => {
        const temp = selFrom.value;
        selFrom.value = selTo.value;
        selTo.value = temp;
        convert();
      });

      inpAmount.addEventListener('input', convert);
      selFrom.addEventListener('change', convert);
      selTo.addEventListener('change', convert);

      convert();
    })();
