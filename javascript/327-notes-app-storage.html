let notes = JSON.parse(localStorage.getItem("lab_sticky_notes") || "null");

    if (!notes) {
      notes = [
        { id: 1, text: "BOM window.location contains href, pathname, and protocol info.", color: "yellow", date: "10:30 AM" },
        { id: 2, text: "Remember to submit Lab Record by Friday 5 PM!", color: "pink", date: "11:15 AM" },
        { id: 3, text: "localStorage allows ~5MB per domain; sessionStorage dies on tab close.", color: "blue", date: "01:45 PM" }
      ];
      localStorage.setItem("lab_sticky_notes", JSON.stringify(notes));
    }

    const board = document.getElementById("notesBoard");

    function renderNotes() {
      board.innerHTML = "";
      if (notes.length === 0) {
        board.innerHTML = `<div style="grid-column: 1/-1; text-align: center; color: #64748b; padding: 2rem;">No sticky notes pinned yet. Create one above!</div>`;
        return;
      }

      notes.forEach(n => {
        const div = document.createElement("div");
        div.className = `sticky-note note-${n.color}`;
        div.innerHTML = `
          <div class="note-text">${n.text}</div>
          <div class="note-footer">
            <span>🕒 ${n.date}</span>
            <button class="btn-del-note" onclick="deleteNote(${n.id})">&times;</button>
          </div>
        `;
        board.appendChild(div);
      });
    }

    function createNote() {
      const textarea = document.getElementById("noteContent");
      const text = textarea.value.trim();
      const color = document.getElementById("noteColor").value;

      if (!text) return;

      const dateStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      notes.push({ id: Date.now(), text, color, date: dateStr });
      localStorage.setItem("lab_sticky_notes", JSON.stringify(notes));

      textarea.value = "";
      renderNotes();
    }

    function deleteNote(id) {
      notes = notes.filter(n => n.id !== id);
      localStorage.setItem("lab_sticky_notes", JSON.stringify(notes));
      renderNotes();
    }

    renderNotes();
