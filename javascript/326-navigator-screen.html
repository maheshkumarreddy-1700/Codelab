Navigator screen

Question: Write a JavaScript program to navigator screen.
▶ Live Output
Working example
JavaScript Code
function renderSpecs() {
      // Navigator
      document.getElementById("navAgent").textContent = navigator.userAgent;
      document.getElementById("navLang").textContent = navigator.language;
      document.getElementById("navPlatform").textContent = navigator.platform || "Win32";
      document.getElementById("navCookie").textContent = navigator.cookieEnabled ? "true (Cookies Allowed)" : "false";
      document.getElementById("navCores").textContent = `${navigator.hardwareConcurrency || 8} CPU Cores`;

      // Screen
      document.getElementById("scrRes").textContent = `${screen.width} × ${screen.height} px`;
      document.getElementById("scrAvail").textContent = `${screen.availWidth} × ${screen.availHeight} px`;
      document.getElementById("scrDepth").textContent = `${screen.colorDepth}-bit color`;
      document.getElementById("viewRes").textContent = `${window.innerWidth} × ${window.innerHeight} px`;
      document.getElementById("scrDpr").textContent = `${window.devicePixelRatio}x (DPR)`;

      // Network
      updateNetworkBadge();
    }

    function updateNetworkBadge() {
      const badge = document.getElementById("onlineBadge");
      if (navigator.onLine) {
        badge.className = "online-indicator online";
        badge.innerHTML = "<span>●</span> <span>ONLINE</span>";
      } else {
        badge.className = "online-indicator offline";
        badge.innerHTML = "<span>●</span> <span>OFFLINE</span>";
      }
    }

    window.addEventListener("online", updateNetworkBadge);
    window.addEventListener("offline", updateNetworkBadge);
    window.addEventListener("resize", renderSpecs);

    renderSpecs();
