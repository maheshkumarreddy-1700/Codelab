let appTasks = JSON.parse(localStorage.getItem("pr_todo_tasks") || "null");
    let currentFilter = "all";

    if (!appTasks) {
      appTasks = [
        { id: 1, title: "Complete HTML & CSS laboratory exercises", priority: "high", completed: true },
        { id: 2, title: "Build JavaScript DOM interactive components", priority: "high", completed: true },
        { id: 3, title: "Study Event Bubbling and Delegation patterns", priority: "med", completed: false }
      ];
      localStorage.setItem("pr_todo_tasks", JSON.stringify(appTasks));
    }

    const taskList = document.getElementById("taskList");
    const pendingCount = document.getElementById("pendingCount");
    const summaryBadge = document.getElementById("summaryBadge");

    function saveAndRender() {
      localStorage.setItem("pr_todo_tasks", JSON.stringify(appTasks));
      render();
    }

    function render() {
      taskList.innerHTML = "";

      let filtered = appTasks;
      if (currentFilter === "active") filtered = appTasks.filter(t => !t.completed);
      if (currentFilter === "completed") filtered = appTasks.filter(t => t.completed);

      const activeTotal = appTasks.filter(t => !t.completed).length;
      pendingCount.textContent = activeTotal;
      summaryBadge.textContent = `${appTasks.length} Tasks Total`;

      if (filtered.length === 0) {
        taskList.innerHTML = `<li style="text-align: center; color: #64748b; padding: 2rem;">No matching tasks found.</li>`;
        return;
      }

      filtered.forEach(t => {
        const li = document.createElement("li");
        li.className = `task-row-card ${t.completed ? 'done' : ''}`;
        li.innerHTML = `
          <div style="display:flex; align-items:center; gap:0.5rem;">
            <input type="checkbox" ${t.completed ? 'checked' : ''} onchange="toggleTask(${t.id})">
            <span class="title">${t.title}</span>
            <span class="priority-badge prio-${t.priority}">${t.priority}</span>
          </div>
          <button style="background:none; border:none; color:#ef4444; cursor:pointer;" onclick="deleteTask(${t.id})">&times;</button>
        `;
        taskList.appendChild(li);
      });
    }

    function createTask() {
      const inp = document.getElementById("inpTaskTitle");
      const prio = document.getElementById("selPriority").value;
      const title = inp.value.trim();

      if (!title) return;

      appTasks.push({ id: Date.now(), title, priority: prio, completed: false });
      inp.value = "";
      saveAndRender();
    }

    function toggleTask(id) {
      const t = appTasks.find(item => item.id === id);
      if (t) t.completed = !t.completed;
      saveAndRender();
    }

    function deleteTask(id) {
      appTasks = appTasks.filter(item => item.id !== id);
      saveAndRender();
    }

    function setFilter(filter, btn) {
      currentFilter = filter;
      document.querySelectorAll(".tab-pill").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      render();
    }

    render();
