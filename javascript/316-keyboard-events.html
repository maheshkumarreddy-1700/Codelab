Keyboard events

Question: Write a JavaScript program to keyboard events.
▶ Live Output
Working example
JavaScript Code
const bigKey = document.getElementById("bigKeyDisplay");
    const valKey = document.getElementById("valKey");
    const valCode = document.getElementById("valCode");
    const valWhich = document.getElementById("valWhich");
    const valLoc = document.getElementById("valLoc");

    const pillCtrl = document.getElementById("pillCtrl");
    const pillShift = document.getElementById("pillShift");
    const pillAlt = document.getElementById("pillAlt");
    const pillMeta = document.getElementById("pillMeta");

    const locNames = ["Standard (0)", "Left (1)", "Right (2)", "Numpad (3)"];

    window.addEventListener("keydown", (e) => {
      // Prevent browser default scrolling on space or arrows during experiment
      if (["Space", "ArrowUp", "ArrowDown"].includes(e.code)) {
        e.preventDefault();
      }

      bigKey.textContent = e.key === " " ? "Space" : e.key;
      valKey.textContent = `"${e.key}"`;
      valCode.textContent = e.code;
      valWhich.textContent = e.which || e.keyCode;
      valLoc.textContent = locNames[e.location] || e.location;

      pillCtrl.classList.toggle("active", e.ctrlKey);
      pillShift.classList.toggle("active", e.shiftKey);
      pillAlt.classList.toggle("active", e.altKey);
      pillMeta.classList.toggle("active", e.metaKey);
    });

    window.addEventListener("keyup", (e) => {
      pillCtrl.classList.toggle("active", e.ctrlKey);
      pillShift.classList.toggle("active", e.shiftKey);
      pillAlt.classList.toggle("active", e.altKey);
      pillMeta.classList.toggle("active", e.metaKey);
    });
