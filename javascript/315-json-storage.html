Json storage

Question: Write a JavaScript program to json storage.

const defaultProfile = {
      name: "Shaik Roshan",
      roll: "22CS042",
      department: "Computer Science & Engineering",
      semester: 6,
      skills: ["HTML5", "CSS3", "JavaScript ES6"],
      enrolled: true
    };

    function syncView() {
      let data = localStorage.getItem("student_profile_json");
      if (!data) {
        localStorage.setItem("student_profile_json", JSON.stringify(defaultProfile, null, 2));
        data = localStorage.getItem("student_profile_json");
      }

      document.getElementById("rawJsonBox").textContent = data;

      try {
        const obj = JSON.parse(data);
        const card = document.getElementById("parsedCard");
        card.innerHTML = `
          <h3 style="color:#f8fafc; margin-bottom:0.25rem;">${obj.name}</h3>
          <p style="color:#38bdf8; font-size:0.85rem; margin-bottom:0.75rem;">Roll: ${obj.roll} • ${obj.department}</p>
          <div style="font-size:0.85rem; color:#cbd5e1; margin-bottom:0.5rem;"><strong>Semester:</strong> ${obj.semester}</div>
          <div style="font-size:0.85rem; color:#cbd5e1; margin-bottom:0.5rem;"><strong>Skills Array:</strong></div>
          <div style="display:flex; flex-wrap:wrap; gap:0.35rem;">
            ${obj.skills.map(s => `<span class="badge badge-primary">${s}</span>`).join('')}
          </div>
        `;
      } catch (e) {
        document.getElementById("parsedCard").textContent = "Parsing Error: " + e.message;
      }
    }

    function addSkillToProfile() {
      const data = JSON.parse(localStorage.getItem("student_profile_json") || "{}");
      const skillsPool = ["React", "Node.js", "MongoDB", "TypeScript", "Docker", "Python"];
      const unused = skillsPool.filter(s => !(data.skills || []).includes(s));

      if (unused.length === 0) {
        alert("All demo skills already added!");
        return;
      }

      data.skills.push(unused[0]);
      localStorage.setItem("student_profile_json", JSON.stringify(data, null, 2));
      syncView();
    }

    function resetProfile() {
      localStorage.setItem("student_profile_json", JSON.stringify(defaultProfile, null, 2));
      syncView();
    }

    syncView();
