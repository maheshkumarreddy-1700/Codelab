Attributes manipulation

Question: Write a JavaScript program to attributes manipulation.
▶ Live Output
Working example
JavaScript Code
const el = document.getElementById("targetElement");

    function renderAttributes() {
      const listing = document.getElementById("attributeListing");
      listing.innerHTML = "";
      
      for (const attr of el.attributes) {
        const row = document.createElement("div");
        row.className = "attr-row";
        row.innerHTML = `
          <span class="attr-name">${attr.name}</span>
          <span class="attr-val">"${attr.value}"</span>
        `;
        listing.appendChild(row);
      }
    }

    function changeAttributes() {
      el.setAttribute("title", "Special Edition Headphones (Updated!)");
      el.setAttribute("data-price", "$299.00");
      el.setAttribute("data-rating", "4.9/5");
      renderAttributes();
    }

    function addCustomData() {
      el.setAttribute("data-status", "in-stock");
      el.setAttribute("data-warehouse", "Bangalore-DC-04");
      renderAttributes();
    }

    function removeCustomAttribute() {
      if (el.hasAttribute("data-category")) {
        el.removeAttribute("data-category");
      } else {
        alert("Attribute 'data-category' has already been removed!");
      }
      renderAttributes();
    }

    renderAttributes();
