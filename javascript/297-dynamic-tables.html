Dynamic tables

Question: Write a JavaScript program to dynamic tables.

function generateTable() {
      const rows = parseInt(document.getElementById("numRows").value) || 3;
      const cols = parseInt(document.getElementById("numCols").value) || 3;
      const pattern = document.getElementById("fillPattern").value;
      const container = document.getElementById("tableContainer");

      container.innerHTML = "";

      const table = document.createElement("table");
      table.className = "generated-table";

      // Table Header
      const thead = table.createTHead();
      const headerRow = thead.insertRow();
      for (let c = 1; c <= cols; c++) {
        const th = document.createElement("th");
        th.textContent = `Column ${c}`;
        headerRow.appendChild(th);
      }

      // Table Body
      const tbody = table.createTBody();
      let counter = 1;

      for (let r = 1; r <= rows; r++) {
        const row = tbody.insertRow();
        for (let c = 1; c <= cols; c++) {
          const cell = row.insertCell();
          if (pattern === "coord") {
            cell.textContent = `(${r}, ${c})`;
          } else if (pattern === "mult") {
            cell.textContent = `${r * c}`;
          } else {
            cell.textContent = `#${counter++}`;
          }
        }
      }

      container.appendChild(table);
    }

    // Initial table render
    generateTable();
