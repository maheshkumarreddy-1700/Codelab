Multi step form

Question: Write a JavaScript program to multi step form.
▶ Live Output
Working example
JavaScript Code
let step = 1;

    const btnPrev = document.getElementById("btnPrev");
    const btnNext = document.getElementById("btnNext");
    const badge = document.getElementById("stepBadge");

    function renderStep() {
      for (let i = 1; i <= 3; i++) {
        const panel = document.getElementById(`panel${i}`);
        const ind = document.getElementById(`step${i}-ind`);
        panel.classList.toggle("active", i === step);
        ind.classList.remove("active", "completed");

        if (i === step) ind.classList.add("active");
        if (i < step) ind.classList.add("completed");
      }

      btnPrev.disabled = (step === 1);
      btnNext.textContent = (step === 3) ? "Complete & Submit" : "Next Step →";
      badge.textContent = `Step ${step} of 3`;

      if (step === 3) {
        document.getElementById("wizSummary").innerHTML = `
          <strong>Account Username:</strong> ${document.getElementById("wizUser").value}<br>
          <strong>Email:</strong> ${document.getElementById("wizEmail").value}<br>
          <strong>Roll Number:</strong> ${document.getElementById("wizRoll").value || "22CS001"}<br>
          <strong>Branch:</strong> ${document.getElementById("wizBranch").value}
        `;
      }
    }

    function nextStep() {
      if (step === 1) {
        if (!document.getElementById("wizUser").value || !document.getElementById("wizEmail").value) {
          alert("Please fill all required account fields!");
          return;
        }
      } else if (step === 2) {
        if (!document.getElementById("wizRoll").value) {
          alert("Please enter roll number!");
          return;
        }
      } else if (step === 3) {
        alert("🎉 Application successfully submitted!");
        step = 1;
        renderStep();
        return;
      }
      step++;
      renderStep();
    }

    function prevStep() {
      if (step > 1) {
        step--;
        renderStep();
      }
    }
