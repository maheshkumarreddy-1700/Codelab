Auto save draft

Question: Write a JavaScript program to auto save draft.
▶ Live Output
Working example
JavaScript Code
let debounceTimer = null;
    const titleInp = document.getElementById("draftTitle");
    const bodyInp = document.getElementById("draftBody");
    const saveStatus = document.getElementById("saveStatus");
    const saveDot = document.getElementById("saveDot");

    function queueAutoSave() {
      saveStatus.textContent = "Saving draft...";
      saveDot.style.color = "#f59e0b";

      clearTimeout(debounceTimer);
      debounceTimer = setTimeout(executeSave, 600);
    }

    function executeSave() {
      const data = {
        title: titleInp.value,
        body: bodyInp.value,
        timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' })
      };

      localStorage.setItem("lab_editor_draft", JSON.stringify(data));
      saveStatus.textContent = `Draft auto-saved at ${data.timestamp}`;
      saveDot.style.color = "#10b981";
    }

    function restoreDraft() {
      const saved = localStorage.getItem("lab_editor_draft");
      if (saved) {
        try {
          const obj = JSON.parse(saved);
          titleInp.value = obj.title || "";
          bodyInp.value = obj.body || "";
          saveStatus.textContent = `Draft restored from ${obj.timestamp || 'previous session'}`;
        } catch (e) {}
      } else {
        titleInp.value = "Experiment 12: JavaScript Web Storage Architecture";
        bodyInp.value = "Web Storage APIs allow web applications to store data locally within the user's browser. Unlike cookies, the storage limit is substantially larger and information is never transferred to the server.";
        executeSave();
      }
    }

    function discardDraft() {
      if (confirm("Discard this draft permanently?")) {
        localStorage.removeItem("lab_editor_draft");
        titleInp.value = "";
        bodyInp.value = "";
        saveStatus.textContent = "Draft discarded";
        saveDot.style.color = "#94a3b8";
      }
    }

    function publishReport() {
      alert("🎉 Assignment report published successfully!");
    }

    restoreDraft();
