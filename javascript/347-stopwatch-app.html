Stopwatch app

Question: Write a JavaScript program to stopwatch app.
▶ Live Output
Working example
JavaScript Code
let isRunning = false;
    let startTime = 0;
    let elapsed = 0;
    let lastLap = 0;
    let timer = null;
    let lapIndex = 0;

    const timeEl = document.getElementById("swTime");
    const msEl = document.getElementById("swMs");
    const btnStart = document.getElementById("btnStart");
    const btnLap = document.getElementById("btnLap");
    const badge = document.getElementById("timerBadge");
    const lapsBody = document.getElementById("lapsBody");
    const ring = document.getElementById("ringBox");

    function formatTime(ms) {
      const m = String(Math.floor(ms / 60000)).padStart(2, '0');
      const s = String(Math.floor((ms % 60000) / 1000)).padStart(2, '0');
      const cs = String(Math.floor((ms % 1000) / 10)).padStart(2, '0');
      return { m, s, cs, full: `${m}:${s}.${cs}` };
    }

    function handleStartStop() {
      if (!isRunning) {
        isRunning = true;
        startTime = Date.now() - elapsed;
        timer = setInterval(() => {
          elapsed = Date.now() - startTime;
          const t = formatTime(elapsed);
          timeEl.textContent = `${t.m}:${t.s}`;
          msEl.textContent = `.${t.cs}`;
        }, 10);

        btnStart.textContent = "Stop";
        btnStart.className = "btn btn-secondary";
        btnStart.style.background = "#ef4444";
        btnStart.style.color = "#fff";
        btnLap.disabled = false;
        badge.textContent = "Running";
        badge.className = "badge badge-primary";
        ring.style.borderTopColor = "#10b981";
      } else {
        isRunning = false;
        clearInterval(timer);
        btnStart.textContent = "Resume";
        btnStart.className = "btn btn-primary";
        btnStart.style.background = "";
        btnLap.disabled = true;
        badge.textContent = "Stopped";
        badge.className = "badge badge-warning";
        ring.style.borderTopColor = "#f59e0b";
      }
    }

    function handleLap() {
      if (!isRunning) return;
      lapIndex++;
      if (lapIndex === 1) lapsBody.innerHTML = "";

      const splitMs = elapsed - lastLap;
      lastLap = elapsed;

      const total = formatTime(elapsed);
      const split = formatTime(splitMs);

      const tr = document.createElement("tr");
      tr.innerHTML = `
        <td><strong>#${lapIndex}</strong></td>
        <td style="color: #38bdf8;">+${split.m}:${split.s}.${split.cs}</td>
        <td>${total.m}:${total.s}.${total.cs}</td>
      `;
      lapsBody.prepend(tr);
    }

    function handleReset() {
      isRunning = false;
      clearInterval(timer);
      elapsed = 0;
      lastLap = 0;
      lapIndex = 0;

      timeEl.textContent = "00:00";
      msEl.textContent = ".00";
      btnStart.textContent = "Start";
      btnStart.className = "btn btn-primary";
      btnStart.style.background = "";
      btnLap.disabled = true;
      badge.textContent = "Ready";
      badge.className = "badge badge-success";
      ring.style.borderTopColor = "#38bdf8";
      lapsBody.innerHTML = `<tr><td colspan="3" style="text-align: center; color: #64748b; padding: 1.5rem;">No laps recorded yet.</td></tr>`;
    }
