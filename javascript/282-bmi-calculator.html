Bmi calculator

Question: Write a JavaScript program to bmi calculator.
▶ Live Output
Working example
JavaScript Code
(function() {
      let currentUnit = 'metric';

      const btnMetric = document.getElementById('btnMetric');
      const btnImperial = document.getElementById('btnImperial');
      const lblWeight = document.getElementById('lblWeight');
      const lblHeight = document.getElementById('lblHeight');
      const inpWeight = document.getElementById('inpWeight');
      const inpHeight = document.getElementById('inpHeight');
      const btnCalc = document.getElementById('btnCalculateBMI');
      const resBMI = document.getElementById('resBMI');
      const resCat = document.getElementById('resCategory');
      const bmiNeedle = document.getElementById('bmiNeedle');
      const idealWeightText = document.getElementById('idealWeightText');

      window.setUnit = function(mode) {
        currentUnit = mode;
        if (mode === 'metric') {
          btnMetric.classList.add('active');
          btnImperial.classList.remove('active');
          lblWeight.textContent = 'Weight (Kilograms):';
          lblHeight.textContent = 'Height (Centimeters):';
          inpWeight.value = '70';
          inpHeight.value = '175';
        } else {
          btnImperial.classList.add('active');
          btnMetric.classList.remove('active');
          lblWeight.textContent = 'Weight (Pounds):';
          lblHeight.textContent = 'Height (Inches):';
          inpWeight.value = '154';
          inpHeight.value = '69';
        }
        calculateBMI();
      };

      function calculateBMI() {
        const weight = parseFloat(inpWeight.value);
        const height = parseFloat(inpHeight.value);

        if (isNaN(weight) || isNaN(height) || weight <= 0 || height <= 0) {
          alert('Please enter valid positive values for weight and height.');
          return;
        }

        let bmi = 0;
        let minIdeal = 0;
        let maxIdeal = 0;

        if (currentUnit === 'metric') {
          const heightM = height / 100;
          bmi = weight / (heightM * heightM);
          minIdeal = (18.5 * heightM * heightM).toFixed(1);
          maxIdeal = (24.9 * heightM * heightM).toFixed(1);
          idealWeightText.innerHTML = `Healthy weight range for your height: <strong>${minIdeal} kg &ndash; ${maxIdeal} kg</strong>`;
        } else {
          bmi = (703 * weight) / (height * height);
          minIdeal = ((18.5 * height * height) / 703).toFixed(1);
          maxIdeal = ((24.9 * height * height) / 703).toFixed(1);
          idealWeightText.innerHTML = `Healthy weight range for your height: <strong>${minIdeal} lbs &ndash; ${maxIdeal} lbs</strong>`;
        }

        resBMI.textContent = bmi.toFixed(1);

        let category = '';
        let color = '';
        let pct = 0;

        if (bmi < 18.5) {
          category = 'Underweight';
          color = '#38bdf8';
          pct = Math.max((bmi / 18.5) * 25, 2);
        } else if (bmi < 25.0) {
          category = 'Normal Healthy Weight';
          color = '#10b981';
          pct = 25 + ((bmi - 18.5) / (24.9 - 18.5)) * 25;
        } else if (bmi < 30.0) {
          category = 'Overweight';
          color = '#f59e0b';
          pct = 50 + ((bmi - 25.0) / (29.9 - 25.0)) * 25;
        } else {
          category = 'Obese';
          color = '#ef4444';
          pct = Math.min(75 + ((bmi - 30.0) / 10.0) * 25, 98);
        }

        resCat.textContent = category;
        resCat.style.color = color;
        bmiNeedle.style.left = `${pct}%`;
      }

      btnCalc.addEventListener('click', calculateBMI);

      // Initial run
      calculateBMI();
    })();
