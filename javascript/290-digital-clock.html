Digital clock

Question: Write a JavaScript program to digital clock.
▶ Live Output
Working example
JavaScript Code
let is24Hour = false;

    const hrsEl = document.getElementById("hours");
    const minsEl = document.getElementById("minutes");
    const secsEl = document.getElementById("seconds");
    const ampmEl = document.getElementById("ampm");
    const dateEl = document.getElementById("dateDisplay");
    const toggleBtn = document.getElementById("modeToggleBtn");

    function tickClock() {
      const now = new Date();

      let h = now.getHours();
      const m = String(now.getMinutes()).padStart(2, '0');
      const s = String(now.getSeconds()).padStart(2, '0');

      let ap = "";
      if (!is24Hour) {
        ap = h >= 12 ? "PM" : "AM";
        h = h % 12 || 12;
        ampmEl.style.display = "inline-block";
      } else {
        ampmEl.style.display = "none";
      }

      hrsEl.textContent = String(h).padStart(2, '0');
      minsEl.textContent = m;
      secsEl.textContent = s;
      ampmEl.textContent = ap;

      dateEl.textContent = now.toLocaleDateString('en-US', {
        weekday: 'long',
        year: 'numeric',
        month: 'long',
        day: 'numeric'
      });
    }

    function toggleHourFormat() {
      is24Hour = !is24Hour;
      toggleBtn.textContent = is24Hour ? "Switch to 12-Hour Format" : "Switch to 24-Hour Format";
      tickClock();
    }

    setInterval(tickClock, 1000);
    tickClock();
