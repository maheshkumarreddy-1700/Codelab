Registration validation

Question: Write a JavaScript program to registration validation.
▶ Live Output
Working example
JavaScript Code
function showError(inputId, errId) {
      const inp = document.getElementById(inputId);
      const err = document.getElementById(errId);
      inp.classList.add("is-invalid");
      inp.classList.remove("is-valid");
      err.style.display = "block";
    }

    function showSuccess(inputId, errId) {
      const inp = document.getElementById(inputId);
      const err = document.getElementById(errId);
      inp.classList.remove("is-invalid");
      inp.classList.add("is-valid");
      err.style.display = "none";
    }

    function validateRegistration(e) {
      e.preventDefault();
      let valid = true;

      // Name
      const name = document.getElementById("regName").value.trim();
      if (!/^[A-Za-z\s]{3,}$/.test(name)) {
        showError("regName", "errName");
        valid = false;
      } else {
        showSuccess("regName", "errName");
      }

      // Email
      const email = document.getElementById("regEmail").value.trim();
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(email)) {
        showError("regEmail", "errEmail");
        valid = false;
      } else {
        showSuccess("regEmail", "errEmail");
      }

      // Phone
      const phone = document.getElementById("regPhone").value.trim();
      if (!/^[6-9]\d{9}$/.test(phone)) {
        showError("regPhone", "errPhone");
        valid = false;
      } else {
        showSuccess("regPhone", "errPhone");
      }

      // Password
      const pwd = document.getElementById("regPassword").value;
      if (pwd.length < 6) {
        showError("regPassword", "errPassword");
        valid = false;
      } else {
        showSuccess("regPassword", "errPassword");
      }

      // Confirm
      const confirm = document.getElementById("regConfirm").value;
      if (pwd !== confirm || confirm === "") {
        showError("regConfirm", "errConfirm");
        valid = false;
      } else {
        showSuccess("regConfirm", "errConfirm");
      }

      if (valid) {
        const box = document.getElementById("successBox");
        box.style.display = "block";
        box.scrollIntoView({ behavior: "smooth" });
      }
    }
