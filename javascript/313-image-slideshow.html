Image slideshow

Question: Write a JavaScript program to image slideshow.

JavaScript Code
const slides = [
      { img: "https://picsum.photos/id/1018/800/450", title: "Mountain Valley Panorama", desc: "Misty mountain peaks under sunrise sky." },
      { img: "https://picsum.photos/id/1015/800/450", title: "River Canyon Rapids", desc: "Glacial river cutting through high granite rocks." },
      { img: "https://picsum.photos/id/1016/800/450", title: "Autumn Pine Wilderness", desc: "Crisp autumn morning in the Pacific Northwest." },
      { img: "https://picsum.photos/id/1019/800/450", title: "Coastal Ocean Mist", desc: "Waves breaking against rugged cliff coastlines." }
    ];

    let current = 0;
    let timer = null;

    const imgEl = document.getElementById("slideImg");
    const titleEl = document.getElementById("slideTitle");
    const descEl = document.getElementById("slideDesc");
    const counterEl = document.getElementById("slideCounter");
    const dotsBox = document.getElementById("dotsContainer");
    const autoBtn = document.getElementById("autoPlayBtn");

    function initDots() {
      dotsBox.innerHTML = "";
      slides.forEach((_, i) => {
        const dot = document.createElement("div");
        dot.className = "dot" + (i === 0 ? " active" : "");
        dot.onclick = () => showSlide(i);
        dotsBox.appendChild(dot);
      });
    }

    function showSlide(idx) {
      if (idx < 0) idx = slides.length - 1;
      if (idx >= slides.length) idx = 0;
      current = idx;

      imgEl.style.opacity = "0.5";
      setTimeout(() => {
        imgEl.src = slides[current].img;
        titleEl.textContent = slides[current].title;
        descEl.textContent = slides[current].desc;
        counterEl.textContent = `${current + 1} / ${slides.length}`;
        
        Array.from(dotsBox.children).forEach((d, i) => {
          d.classList.toggle("active", i === current);
        });
        imgEl.style.opacity = "1";
      }, 150);
    }

    function nextSlide() { showSlide(current + 1); }
    function prevSlide() { showSlide(current - 1); }

    function startAutoPlay() {
      timer = setInterval(nextSlide, 3500);
      autoBtn.textContent = "⏸️ Pause";
    }

    function stopAutoPlay() {
      clearInterval(timer);
      timer = null;
      autoBtn.textContent = "▶️ Play";
    }

    function toggleAutoPlay() {
      if (timer) stopAutoPlay();
      else startAutoPlay();
    }

    initDots();
    startAutoPlay();
