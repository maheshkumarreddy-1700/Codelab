Salary calculator

Question: Write a JavaScript program to salary calculator.
▶ Live Output
Working example
JavaScript Code
function calculatePayroll() {
      const basic = parseFloat(document.getElementById("empBasic").value) || 0;
      const daPct = parseFloat(document.getElementById("empDa").value) || 0;
      const hraPct = parseFloat(document.getElementById("empHra").value) || 0;
      const bonus = parseFloat(document.getElementById("empBonus").value) || 0;

      const da = (basic * daPct) / 100;
      const hra = (basic * hraPct) / 100;
      const gross = basic + da + hra + bonus;

      const pf = (basic * 12) / 100;
      const pt = 200;
      const net = gross - (pf + pt);

      const fmt = (n) => "₹" + Math.round(n).toLocaleString('en-IN');

      document.getElementById("slipBasic").textContent = fmt(basic);
      document.getElementById("slipDa").textContent = fmt(da);
      document.getElementById("slipHra").textContent = fmt(hra);
      document.getElementById("slipBonus").textContent = fmt(bonus);
      document.getElementById("slipGross").textContent = fmt(gross);
      document.getElementById("slipPf").textContent = "- " + fmt(pf);
      document.getElementById("slipPt").textContent = "- " + fmt(pt);
      document.getElementById("slipNet").textContent = fmt(net);
    }

    calculatePayroll();
