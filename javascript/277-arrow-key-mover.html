Arrow key mover

Question: Write a JavaScript program to arrow key mover.
▶ Live Output
Working example
JavaScript Code
let posX = 150, posY = 150;
    let score = 0;
    const STEP = 25;
    const avatar = document.getElementById("avatar");
    const coin = document.getElementById("coin");
    const arena = document.getElementById("gameArena");
    const scoreVal = document.getElementById("scoreVal");
    const xPosVal = document.getElementById("xPosVal");
    const yPosVal = document.getElementById("yPosVal");

    function spawnCoin() {
      const maxX = arena.clientWidth - 40;
      const maxY = arena.clientHeight - 40;
      const randX = Math.floor(Math.random() * (maxX - 20) + 20);
      const randY = Math.floor(Math.random() * (maxY - 20) + 20);
      coin.style.left = `${randX}px`;
      coin.style.top = `${randY}px`;
    }

    function checkCollision() {
      const aRect = avatar.getBoundingClientRect();
      const cRect = coin.getBoundingClientRect();

      if (!(aRect.right < cRect.left || 
            aRect.left > cRect.right || 
            aRect.bottom < cRect.top || 
            aRect.top > cRect.bottom)) {
        score += 10;
        scoreVal.textContent = score;
        spawnCoin();
      }
    }

    function update() {
      const maxX = arena.clientWidth - 44;
      const maxY = arena.clientHeight - 44;

      posX = Math.max(0, Math.min(maxX, posX));
      posY = Math.max(0, Math.min(maxY, posY));

      avatar.style.left = `${posX}px`;
      avatar.style.top = `${posY}px`;
      xPosVal.textContent = posX;
      yPosVal.textContent = posY;

      checkCollision();
    }

    function move(dir) {
      if (dir === 'up') posY -= STEP;
      if (dir === 'down') posY += STEP;
      if (dir === 'left') posX -= STEP;
      if (dir === 'right') posX += STEP;
      update();
    }

    window.addEventListener("keydown", (e) => {
      if (["ArrowUp", "KeyW", "w"].includes(e.key)) { e.preventDefault(); move('up'); }
      if (["ArrowDown", "KeyS", "s"].includes(e.key)) { e.preventDefault(); move('down'); }
      if (["ArrowLeft", "KeyA", "a"].includes(e.key)) { e.preventDefault(); move('left'); }
      if (["ArrowRight", "KeyD", "d"].includes(e.key)) { e.preventDefault(); move('right'); }
    });
