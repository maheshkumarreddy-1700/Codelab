Create a counter application with increment and decrement buttons

Question: Write a JavaScript program to create a counter application with increment and decrement buttons.
▶ Live Output
Working example
JavaScript Code
function check(){
    let answer=document.querySelector('input[name="answer"]:checked');

    if(!answer){
        result("Select an answer.");
        return;
    }

    result(answer.value==="CSS" ? "Correct! 🎉" : "Wrong answer.");
}

function result(text){
    document.getElementById("result").textContent=text;
}
