Text innerhtml

Question: Write a JavaScript program to text innerhtml.
▶ Live Output
Working example
JavaScript Code
const initialHTML = document.getElementById("sampleNode").innerHTML;

    function readProperties() {
      const el = document.getElementById("sampleNode");
      document.getElementById("outTextContent").textContent = el.textContent;
      document.getElementById("outInnerText").textContent = el.innerText;
      document.getElementById("outInnerHTML").textContent = el.innerHTML;
    }

    function setAsText() {
      const el = document.getElementById("sampleNode");
      el.textContent = "Assigned via textContent: <strong>Tags are NOT parsed</strong> and displayed as plain text!";
      readProperties();
    }

    function setAsHtml() {
      const el = document.getElementById("sampleNode");
      el.innerHTML = "Assigned via innerHTML: <mark style='background:#f59e0b; color:#000; padding:2px 6px; border-radius:4px;'>Parsed HTML markup!</mark> <button style='padding:4px 8px;'>Active Button</button>";
      readProperties();
    }

    function resetSample() {
      document.getElementById("sampleNode").innerHTML = initialHTML;
      readProperties();
    }

    // Run initial read
    readProperties();
