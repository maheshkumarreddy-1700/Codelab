Fibonacci

Question: Write a JavaScript program to fibonacci.
▶ Live Output

JavaScript Code
function generateFibonacci() {
      const n = parseInt(document.getElementById('fibo-terms').value);
      const container = document.getElementById('res-fibo');
      if (isNaN(n) || n < 1) {
        container.innerHTML = `<span style="color: #ef4444;">Please enter a positive integer (N &gt;= 1).</span>`;
        return;
      }
      let series = [0];
      if (n > 1) {
        series.push(1);
        for (let i = 2; i < n; i++) {
          series.push(series[i - 1] + series[i - 2]);
        }
      }
      container.innerHTML = series.map((num, idx) => `
        <span class="fibo-pill">
          <small style="opacity: 0.6; font-size: 0.7rem; display: block;">T${idx+1}</small>
          ${num.toLocaleString()}
        </span>
      `).join('');
    }
    generateFibonacci();
