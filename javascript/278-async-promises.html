Async promises

Question: Write a JavaScript program to async promises.
▶ Live Output
Working example
JavaScript Code
function simulateFetch(succeed) {
      return new Promise((resolve, reject) => {
        setTimeout(() => {
          if (succeed) {
            resolve({ rollNo: "23CS01", name: "Shaik Roshan", status: "Active Student", recordsVerified: 150 });
          } else {
            reject(new Error("404 Remote Academic Record Not Found"));
          }
        }, 1200);
      });
    }

    async function runAsyncSuccess() {
      const res = document.getElementById('res-async');
      res.innerHTML = `[${new Date().toLocaleTimeString()}] ⏳ [PENDING] Dispatching async request to endpoint...<br>`;
      try {
        const payload = await simulateFetch(true);
        res.innerHTML += `[${new Date().toLocaleTimeString()}] <span style="color: #16a34a; font-weight: 700;">✅ [RESOLVED] Successfully fetched data:</span><br>` +
          `&gt; Student: ${payload.name} (${payload.rollNo})<br>&gt; Status: ${payload.status} (${payload.recordsVerified} experiments verified)`;
      } catch (err) {
        res.innerHTML += `[${new Date().toLocaleTimeString()}] ❌ [REJECTED] ${err.message}`;
      }
    }

    async function runAsyncFailure() {
      const res = document.getElementById('res-async');
      res.innerHTML = `[${new Date().toLocaleTimeString()}] ⏳ [PENDING] Dispatching async request...<br>`;
      try {
        await simulateFetch(false);
      } catch (err) {
        res.innerHTML += `[${new Date().toLocaleTimeString()}] <span style="color: #ef4444; font-weight: 700;">❌ [REJECTED] Handled in catch block:</span> ${err.message}`;
      }
    }
