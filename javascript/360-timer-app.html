Timer app

Question: Write a JavaScript program to timer app.
▶ Live Output
Working example
JavaScript Code
let totalSecs = 25 * 60;
    let initialSecs = 25 * 60;
    let isRunning = false;
    let timer = null;
    let cycles = 0;

    const display = document.getElementById("pomoDisplay");
    const btnStart = document.getElementById("btnPomoStart");
    const badge = document.getElementById("sessionBadge");
    const completedEl = document.getElementById("completedCount");

    function renderDisplay() {
      const m = String(Math.floor(totalSecs / 60)).padStart(2, '0');
      const s = String(totalSecs % 60).padStart(2, '0');
      display.textContent = `${m}:${s}`;
    }

    function setMode(mins, label) {
      clearInterval(timer);
      isRunning = false;
      initialSecs = mins * 60;
      totalSecs = initialSecs;
      badge.textContent = label;
      btnStart.textContent = "Start " + (mins === 25 ? "Focus" : "Break");

      document.querySelectorAll(".pomo-mode-btn").forEach(b => {
        b.classList.toggle("active", b.textContent.includes(`${mins}m`));
      });

      renderDisplay();
    }

    function togglePomo() {
      if (!isRunning) {
        isRunning = true;
        btnStart.textContent = "Pause";
        btnStart.className = "btn btn-secondary";

        timer = setInterval(() => {
          if (totalSecs <= 0) {
            clearInterval(timer);
            isRunning = false;
            btnStart.textContent = "Start";
            btnStart.className = "btn btn-primary";
            if (initialSecs === 25 * 60) {
              cycles++;
              completedEl.textContent = cycles;
              alert("🎉 Excellent focus! 25-minute Pomodoro finished. Enjoy a 5-minute break!");
              setMode(5, "Short Break");
            } else {
              alert("☕ Break time over! Back to focused study.");
              setMode(25, "Work Session");
            }
            return;
          }
          totalSecs--;
          renderDisplay();
        }, 1000);
      } else {
        clearInterval(timer);
        isRunning = false;
        btnStart.textContent = "Resume";
        btnStart.className = "btn btn-primary";
      }
    }

    function resetPomo() {
      clearInterval(timer);
      isRunning = false;
      totalSecs = initialSecs;
      btnStart.textContent = "Start";
      btnStart.className = "btn btn-primary";
      renderDisplay();
    }

    renderDisplay();
