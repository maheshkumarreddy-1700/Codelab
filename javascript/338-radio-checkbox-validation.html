Radio checkbox validation

Question: Write a JavaScript program to radio checkbox validation.
▶ Live Output
Working example
JavaScript Code
function validateSelections(e) {
      e.preventDefault();
      let valid = true;

      // 1. Radio check
      const radio = document.querySelector('input[name="track"]:checked');
      const errTrack = document.getElementById("errTrack");
      if (!radio) {
        errTrack.style.display = "block";
        valid = false;
      } else {
        errTrack.style.display = "none";
      }

      // 2. Checkboxes check (min 2)
      const skills = document.querySelectorAll('input[name="skills"]:checked');
      const errSkills = document.getElementById("errSkills");
      if (skills.length < 2) {
        errSkills.style.display = "block";
        valid = false;
      } else {
        errSkills.style.display = "none";
      }

      // 3. Terms
      const terms = document.getElementById("termsCheck");
      const errTerms = document.getElementById("errTerms");
      if (!terms.checked) {
        errTerms.style.display = "block";
        valid = false;
      } else {
        errTerms.style.display = "none";
      }

      if (valid) {
        const skillList = Array.from(skills).map(s => s.value).join(", ");
        alert(`✔ Selections Verified!\nTrack: ${radio.value}\nLanguages: ${skillList}`);
      }
    }
