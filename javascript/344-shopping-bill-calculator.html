Shopping bill calculator

Question: Write a JavaScript program to shopping bill calculator.
▶ Live Output
Working example
JavaScript Code
function updateInvoice() {
      const q1 = Math.max(0, parseInt(document.getElementById("qty1").value) || 0);
      const q2 = Math.max(0, parseInt(document.getElementById("qty2").value) || 0);
      const q3 = Math.max(0, parseInt(document.getElementById("qty3").value) || 0);

      const t1 = q1 * 2499;
      const t2 = q2 * 899;
      const t3 = q3 * 399;

      const subtotal = t1 + t2 + t3;
      const discount = Math.round(subtotal * 0.10);
      const taxable = subtotal - discount;
      const gst = Math.round(taxable * 0.18);
      const grandTotal = taxable + gst;

      const fmt = (n) => "₹" + n.toLocaleString('en-IN');

      document.getElementById("line1").textContent = fmt(t1);
      document.getElementById("line2").textContent = fmt(t2);
      document.getElementById("line3").textContent = fmt(t3);

      document.getElementById("subTotal").textContent = fmt(subtotal);
      document.getElementById("discountVal").textContent = "- " + fmt(discount);
      document.getElementById("taxVal").textContent = "+ " + fmt(gst);
      document.getElementById("grandTotal").textContent = fmt(grandTotal);
    }

    updateInvoice();
