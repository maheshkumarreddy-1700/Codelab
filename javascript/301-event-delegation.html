Event delegation

Question: Write a JavaScript program to event delegation.
▶ Live Output
Working example
JavaScript Code
const parentList = document.getElementById("delegatedList");
    const input = document.getElementById("newItemTitle");
    let nextId = 103;

    // Single Delegated Listener
    parentList.addEventListener("click", (e) => {
      const action = e.target.getAttribute("data-action");
      const row = e.target.closest(".list-item-btn");

      if (!action || !row) return;

      const itemId = row.getAttribute("data-id");
      const itemTitle = row.querySelector("span:not(.item-badge)").textContent;

      if (action === "delete") {
        row.remove();
      } else if (action === "view") {
        alert(`Details for Item ID: ${itemId}\nTitle: "${itemTitle}"`);
      }
    });

    function addNewItem() {
      const title = input.value.trim();
      if (!title) return;

      const li = document.createElement("li");
      li.className = "list-item-btn";
      li.setAttribute("data-id", nextId);
      li.innerHTML = `
        <div>
          <span class="item-badge">ID: ${nextId++}</span>
          <span>${title}</span>
        </div>
        <div style="display: flex; gap: 0.35rem;">
          <button class="btn-action-view" data-action="view">View</button>
          <button class="btn-action-del" data-action="delete">Delete</button>
        </div>
      `;
      parentList.appendChild(li);
      input.value = "";
    }
