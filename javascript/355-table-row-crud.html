Table row crud

Question: Write a JavaScript program to table row crud.
▶ Live Output
Working example
JavaScript Code
let students = [
      { roll: "22CSE01", name: "Shaik Roshan", dept: "CSE", marks: 96 },
      { roll: "22CSE05", name: "Kiran Kumar", dept: "IT", marks: 84 },
      { roll: "22ECE12", name: "Pooja Hegde", dept: "ECE", marks: 91 }
    ];

    function renderTable() {
      const tbody = document.getElementById("tableBody");
      tbody.innerHTML = "";

      if (students.length === 0) {
        tbody.innerHTML = `<tr><td colspan="6" style="text-align: center; color: #94a3b8; padding: 2rem;">No records found. Add one above!</td></tr>`;
        return;
      }

      students.forEach((s, idx) => {
        const row = tbody.insertRow();
        row.innerHTML = `
          <td>${idx + 1}</td>
          <td><strong>${s.roll}</strong></td>
          <td>${s.name}</td>
          <td><span class="badge badge-primary">${s.dept}</span></td>
          <td>${s.marks}%</td>
          <td>
            <div class="action-btn-group">
              <button class="btn-edit" onclick="editRecord(${idx})">Edit</button>
              <button class="btn-del" onclick="deleteRecord(${idx})">Delete</button>
            </div>
          </td>
        `;
      });
    }

    function handleSaveRecord() {
      const name = document.getElementById("studName").value.trim();
      const roll = document.getElementById("studRoll").value.trim();
      const dept = document.getElementById("studDept").value;
      const marks = document.getElementById("studMarks").value.trim();
      const editIdx = parseInt(document.getElementById("editIndex").value);

      if (!name || !roll || !marks) {
        alert("Please fill all required fields!");
        return;
      }

      if (editIdx >= 0) {
        students[editIdx] = { name, roll, dept, marks };
        document.getElementById("editIndex").value = -1;
        document.getElementById("submitBtn").textContent = "Add Record";
      } else {
        students.push({ name, roll, dept, marks });
      }

      document.getElementById("studName").value = "";
      document.getElementById("studRoll").value = "";
      document.getElementById("studMarks").value = "";
      renderTable();
    }

    function editRecord(idx) {
      const s = students[idx];
      document.getElementById("studName").value = s.name;
      document.getElementById("studRoll").value = s.roll;
      document.getElementById("studDept").value = s.dept;
      document.getElementById("studMarks").value = s.marks;
      document.getElementById("editIndex").value = idx;
      document.getElementById("submitBtn").textContent = "Update Record";
    }

    function deleteRecord(idx) {
      if (confirm(`Are you sure you want to delete ${students[idx].name}?`)) {
        students.splice(idx, 1);
        renderTable();
      }
    }

    renderTable();
