Dom classes

Question: Write a JavaScript program to dom classes.
▶ Live Output
Working example
JavaScript Code
const box = document.getElementById("targetBox");
    const classDisplay = document.getElementById("classDisplay");

    function syncDisplay() {
      classDisplay.textContent = box.className;
    }

    function toggleHighlight() {
      box.classList.toggle("theme-highlight");
      syncDisplay();
    }

    function togglePill() {
      box.classList.toggle("shape-pill");
      syncDisplay();
    }

    function toggleActive() {
      box.classList.toggle("state-active");
      syncDisplay();
    }

    function toggleGlow() {
      box.classList.toggle("text-glow");
      syncDisplay();
    }

    function checkActive() {
      const has = box.classList.contains("theme-highlight");
      alert("box.classList.contains('theme-highlight') => " + has);
    }

    function resetClasses() {
      box.className = "sample-box";
      syncDisplay();
    }
