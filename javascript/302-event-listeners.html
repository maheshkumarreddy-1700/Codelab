Event listeners

Question: Write a JavaScript program to event listeners.

JavaScript Code
const targetBtn = document.getElementById("targetBtn");
    const historyLog = document.getElementById("historyLog");
    const statusEl = document.getElementById("listenerStatus");
    const btnH1 = document.getElementById("btnToggleH1");
    const btnH2 = document.getElementById("btnToggleH2");

    let h1Attached = true;
    let h2Attached = true;
    let clickCount = 0;

    function logHistory(msg) {
      const div = document.createElement("div");
      div.textContent = `> ${msg}`;
      historyLog.prepend(div);
    }

    function clearHistory() {
      historyLog.innerHTML = "";
    }

    // Handler 1
    function flashEffect() {
      targetBtn.style.borderColor = "#38bdf8";
      targetBtn.style.boxShadow = "0 0 20px rgba(56, 189, 248, 0.6)";
      setTimeout(() => {
        targetBtn.style.borderColor = "#6366f1";
        targetBtn.style.boxShadow = "";
      }, 200);
      logHistory("[H1] Flash & Highlight Triggered");
    }

    // Handler 2
    function incrementCounter() {
      clickCount++;
      logHistory(`[H2] Click Counter Incremented to ${clickCount}`);
    }

    function updateStatus() {
      let count = (h1Attached ? 1 : 0) + (h2Attached ? 1 : 0);
      statusEl.textContent = `${count} Active`;
      statusEl.className = count > 0 ? "status-badge status-active" : "status-badge status-inactive";
    }

    function toggleHandler1() {
      if (h1Attached) {
        targetBtn.removeEventListener("click", flashEffect);
        h1Attached = false;
        btnH1.textContent = "Attach";
        btnH1.className = "btn btn-primary";
        logHistory("Detached Handler 1 via removeEventListener()");
      } else {
        targetBtn.addEventListener("click", flashEffect);
        h1Attached = true;
        btnH1.textContent = "Remove";
        btnH1.className = "btn btn-secondary";
        logHistory("Attached Handler 1 via addEventListener()");
      }
      updateStatus();
    }

    function toggleHandler2() {
      if (h2Attached) {
        targetBtn.removeEventListener("click", incrementCounter);
        h2Attached = false;
        btnH2.textContent = "Attach";
        btnH2.className = "btn btn-primary";
        logHistory("Detached Handler 2 via removeEventListener()");
      } else {
        targetBtn.addEventListener("click", incrementCounter);
        h2Attached = true;
        btnH2.textContent = "Remove";
        btnH2.className = "btn btn-secondary";
        logHistory("Attached Handler 2 via addEventListener()");
      }
      updateStatus();
    }

    function attachOnceHandler() {
      targetBtn.addEventListener("click", () => {
        alert("🎉 Special VIP One-Time Bonus Awarded! (Will never fire again on subsequent clicks)");
        logHistory("[VIP Once] Fired and automatically removed from event pool.");
      }, { once: true });
      logHistory("Attached { once: true } listener successfully.");
    }

    // Initialize initial handlers
    targetBtn.addEventListener("click", flashEffect);
    targetBtn.addEventListener("click", incrementCounter);
    updateStatus();
