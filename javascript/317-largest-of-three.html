Largest of three

Question: Write a JavaScript program to largest of three.
▶ Live Output
Working example
JavaScript Code
function findLargest() {
      const a = parseFloat(document.getElementById('valA').value);
      const b = parseFloat(document.getElementById('valB').value);
      const c = parseFloat(document.getElementById('valC').value);

      if (isNaN(a) || isNaN(b) || isNaN(c)) {
        document.getElementById('res-largest').textContent = "Please enter valid numbers in all 3 fields.";
        return;
      }

      let winner = "";
      let max = a;

      if (a >= b && a >= c) {
        max = a; winner = "Number A";
      } else if (b >= a && b >= c) {
        max = b; winner = "Number B";
      } else {
        max = c; winner = "Number C";
      }

      document.getElementById('res-largest').innerHTML = `
        The largest value is <strong>${max}</strong> (${winner}).<br>
        <small style="color: #64748b;">Math.max(${a}, ${b}, ${c}) = ${Math.max(a, b, c)}</small>
      `;
    }
