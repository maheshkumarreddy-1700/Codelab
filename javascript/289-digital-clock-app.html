Digital clock app

Question: Write a JavaScript program to digital clock app.
▶ Live Output
Working example
JavaScript Code
function updateWorldClocks() {
      const zones = [
        { id: "Ist", tz: "Asia/Kolkata" },
        { id: "London", tz: "Europe/London" },
        { id: "Ny", tz: "America/New_York" },
        { id: "Tokyo", tz: "Asia/Tokyo" }
      ];

      const now = new Date();

      zones.forEach(z => {
        const timeStr = now.toLocaleTimeString('en-US', {
          timeZone: z.tz,
          hour12: true,
          hour: '2-digit',
          minute: '2-digit',
          second: '2-digit'
        });

        const dateStr = now.toLocaleDateString('en-US', {
          timeZone: z.tz,
          weekday: 'short',
          month: 'short',
          day: 'numeric',
          year: 'numeric'
        });

        document.getElementById(`time${z.id}`).textContent = timeStr;
        document.getElementById(`date${z.id}`).textContent = dateStr;
      });
    }

    setInterval(updateWorldClocks, 1000);
    updateWorldClocks();
