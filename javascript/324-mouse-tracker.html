Mouse tracker

Question: Write a JavaScript program to mouse tracker.
▶ Live Output
Working example
JavaScript Code
const radar = document.getElementById("radarBox");
    const spotlight = document.getElementById("spotlight");
    const crossH = document.getElementById("crossH");
    const crossV = document.getElementById("crossV");

    const offsetCoord = document.getElementById("offsetCoord");
    const clientCoord = document.getElementById("clientCoord");
    const pageCoord = document.getElementById("pageCoord");
    const screenCoord = document.getElementById("screenCoord");

    radar.addEventListener("mousemove", (e) => {
      const ox = e.offsetX;
      const oy = e.offsetY;

      spotlight.style.left = `${ox}px`;
      spotlight.style.top = `${oy}px`;
      crossH.style.top = `${oy}px`;
      crossV.style.left = `${ox}px`;

      offsetCoord.textContent = `${ox}, ${oy}`;
      clientCoord.textContent = `${e.clientX}, ${e.clientY}`;
      pageCoord.textContent = `${e.pageX}, ${e.pageY}`;
      screenCoord.textContent = `${e.screenX}, ${e.screenY}`;
    });
