Localstorage crud

Question: Write a JavaScript program to localstorage crud.

// Initialize sample data if empty
    if (!localStorage.getItem("student_name")) {
      localStorage.setItem("student_name", "Shaik Roshan");
      localStorage.setItem("app_theme", "dark_cyber");
      localStorage.setItem("lab_subject", "Web Technologies Lab");
    }

    function refreshTable() {
      const tbody = document.getElementById("storageTbody");
      const badge = document.getElementById("countBadge");
      tbody.innerHTML = "";

      const count = localStorage.length;
      badge.textContent = `${count} Key${count === 1 ? '' : 's'} Stored`;

      if (count === 0) {
        tbody.innerHTML = `<tr><td colspan="4" style="text-align:center; color:#64748b; padding:2rem;">Storage is empty. Add a key on the left!</td></tr>`;
        return;
      }

      for (let i = 0; i < count; i++) {
        const key = localStorage.key(i);
        const val = localStorage.getItem(key);
        const tr = document.createElement("tr");
        tr.innerHTML = `
          <td>${i + 1}</td>
          <td style="color: #38bdf8;"><strong>${key}</strong></td>
          <td>${val}</td>
          <td><button class="btn-del-key" onclick="deleteKey('${key}')">Delete</button></td>
        `;
        tbody.appendChild(tr);
      }
    }

    function saveItem() {
      const k = document.getElementById("storageKey").value.trim();
      const v = document.getElementById("storageVal").value.trim();
      if (!k) {
        alert("Please enter a key name!");
        return;
      }
      localStorage.setItem(k, v);
      document.getElementById("storageKey").value = "";
      document.getElementById("storageVal").value = "";
      refreshTable();
    }

    function readItem() {
      const k = document.getElementById("storageKey").value.trim();
      if (!k) {
        alert("Enter key name to query!");
        return;
      }
      const val = localStorage.getItem(k);
      if (val !== null) {
        alert(`Found key '${k}':\n"${val}"`);
      } else {
        alert(`Key '${k}' does not exist in localStorage.`);
      }
    }

    function deleteKey(key) {
      localStorage.removeItem(key);
      refreshTable();
    }

    function clearAllStorage() {
      if (confirm("Are you sure you want to wipe all localStorage data?")) {
        localStorage.clear();
        refreshTable();
      }
    }

    refreshTable();
