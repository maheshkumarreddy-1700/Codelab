Visitor counter

Question: Write a JavaScript program to visitor counter.
▶ Live Output
Working example
JavaScript Code
function updateAnalytics() {
      let count = parseInt(localStorage.getItem("lab_visit_counter") || "0");
      count++;
      localStorage.setItem("lab_visit_counter", count);

      const now = new Date().toLocaleString();
      let first = localStorage.getItem("lab_first_visit");
      if (!first) {
        first = now;
        localStorage.setItem("lab_first_visit", first);
      }

      const prev = localStorage.getItem("lab_last_visit") || "First time opening this page!";
      localStorage.setItem("lab_last_visit", now);

      document.getElementById("visitCountDisplay").textContent = count;
      document.getElementById("firstVisit").textContent = first;
      document.getElementById("prevVisit").textContent = prev;
      document.getElementById("currentSession").textContent = now;
    }

    function simulateReload() {
      updateAnalytics();
    }

    function resetAnalytics() {
      localStorage.removeItem("lab_visit_counter");
      localStorage.removeItem("lab_first_visit");
      localStorage.removeItem("lab_last_visit");
      updateAnalytics();
    }

    updateAnalytics();
