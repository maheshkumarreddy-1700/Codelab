Event bubbling capturing

Question: Write a JavaScript program to event bubbling capturing.
▶ Live Output
Working example
JavaScript Code
const child = document.getElementById("child");
    const parent = document.getElementById("parent");
    const grandparent = document.getElementById("grandparent");
    const logBox = document.getElementById("propagationLog");
    const stopCheck = document.getElementById("stopPropCheck");

    function logPropagation(phase, elId) {
      const item = document.createElement("div");
      item.className = `log-item item-${phase.toLowerCase()}`;
      item.textContent = `[${phase}] Fired on #${elId}`;
      logBox.appendChild(item);
      logBox.scrollTop = logBox.scrollHeight;
    }

    function clearPropagationLog() {
      logBox.innerHTML = "";
    }

    function handleEvent(phase, elId, event) {
      logPropagation(phase, elId);
      if (stopCheck.checked && elId === "child") {
        event.stopPropagation();
        const haltNotice = document.createElement("div");
        haltNotice.style.color = "#f43f5e";
        haltNotice.style.fontWeight = "bold";
        haltNotice.textContent = "🛑 event.stopPropagation() called! Propagation halted.";
        logBox.appendChild(haltNotice);
      }
    }

    // Capture listeners (true)
    grandparent.addEventListener("click", (e) => handleEvent("Capture", "grandparent", e), true);
    parent.addEventListener("click", (e) => handleEvent("Capture", "parent", e), true);
    child.addEventListener("click", (e) => handleEvent("Capture", "child", e), true);

    // Bubble listeners (false)
    child.addEventListener("click", (e) => handleEvent("Bubble", "child", e), false);
    parent.addEventListener("click", (e) => handleEvent("Bubble", "parent", e), false);
    grandparent.addEventListener("click", (e) => handleEvent("Bubble", "grandparent", e), false);
