Font size preference

Question: Write a JavaScript program to font size preference.
▶ Live Output
Working example
JavaScript Code
const article = document.getElementById("articleContent");
    const badge = document.getElementById("sizeBadge");

    function setFontSize(size) {
      article.style.fontSize = `${size}px`;
      badge.textContent = `Size: ${size}px`;
      localStorage.setItem("lab_font_size_pref", size);
    }

    const savedSize = localStorage.getItem("lab_font_size_pref") || 18;
    setFontSize(parseInt(savedSize));
