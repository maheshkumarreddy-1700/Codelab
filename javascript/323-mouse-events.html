Mouse events

Question: Write a JavaScript program to mouse events.
▶ Live Output
Working example
JavaScript Code
const pad = document.getElementById("mousePad");
    const logBox = document.getElementById("eventLog");
    const currentText = document.getElementById("currentEventText");
    const btnVal = document.getElementById("mouseBtnVal");
    const posX = document.getElementById("posXVal");
    const posY = document.getElementById("posYVal");

    function logEvent(type, detail) {
      currentText.textContent = `Event: ${type}`;
      const entry = document.createElement("div");
      entry.className = "log-entry";
      entry.innerHTML = `<span class="log-name">${type}</span><span class="log-detail">${detail}</span>`;
      logBox.prepend(entry);
    }

    function clearLog() {
      logBox.innerHTML = "";
    }

    pad.addEventListener("click", (e) => {
      btnVal.textContent = "Left Click (0)";
      logEvent("click", `Coordinates: (${e.offsetX}, ${e.offsetY})`);
    });

    pad.addEventListener("dblclick", () => {
      logEvent("dblclick", "Rapid Double Click Triggered!");
    });

    pad.addEventListener("contextmenu", (e) => {
      e.preventDefault();
      btnVal.textContent = "Right Click (2)";
      logEvent("contextmenu", "Right Click (Default Menu Suppressed)");
    });

    pad.addEventListener("mouseenter", () => logEvent("mouseenter", "Mouse pointer entered"));
    pad.addEventListener("mouseleave", () => {
      logEvent("mouseleave", "Mouse pointer exited");
      btnVal.textContent = "None";
    });

    pad.addEventListener("mousedown", () => logEvent("mousedown", "Button pressed DOWN"));
    pad.addEventListener("mouseup", () => logEvent("mouseup", "Button released UP"));

    pad.addEventListener("mousemove", (e) => {
      posX.textContent = e.offsetX;
      posY.textContent = e.offsetY;
    });
