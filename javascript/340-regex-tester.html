Question: Write a JavaScript program to regex tester.

function evaluateRegEx() {
      const pattern = document.getElementById("regPattern").value;
      const flags = document.getElementById("regFlags").value;
      const text = document.getElementById("regTestText").value;
      const preview = document.getElementById("highlightPreview");
      const badge = document.getElementById("matchBadge");

      if (!pattern) {
        preview.textContent = text;
        badge.textContent = "0 Matches";
        return;
      }

      try {
        const re = new RegExp(pattern, flags);
        const matches = text.match(re);
        const count = matches ? matches.length : 0;
        badge.textContent = `${count} Match${count === 1 ? '' : 'es'}`;

        const escaped = text
          .replace(/&/g, "&amp;")
          .replace(/</g, "&lt;")
          .replace(/>/g, "&gt;");

        const highlighted = escaped.replace(re, (m) => `<mark class="match-highlight">${m}</mark>`);
        preview.innerHTML = highlighted;
      } catch (e) {
        badge.textContent = "Syntax Error";
        preview.innerHTML = `<span style="color:#ef4444;">Error: ${e.message}</span>`;
      }
    }

    evaluateRegEx();
