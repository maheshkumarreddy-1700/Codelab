Reverse number

Question: Write a JavaScript program to reverse number.
▶ Live Output
Working example
JavaScript Code
function reverseInput() {
      const val = document.getElementById('rev-input').value.trim();
      const res = document.getElementById('res-reverse');
      if (!val) {
        res.textContent = "Please enter an input.";
        return;
      }
      const num = Number(val);
      if (!isNaN(num) && Number.isInteger(num)) {
        let isNeg = num < 0;
        let n = Math.abs(num);
        let rev = 0;
        while (n > 0) {
          rev = (rev * 10) + (n % 10);
          n = Math.floor(n / 10);
        }
        const finalNum = isNeg ? -rev : rev;
        res.innerHTML = `Original Integer: <strong>${val}</strong> &rarr; Reversed Integer: <strong>${finalNum}</strong> <br><small style="color: #64748b;">(Computed purely via while loop arithmetic)</small>`;
      } else {
        const revStr = val.split('').reverse().join('');
        res.innerHTML = `Original String: <strong>"${val}"</strong> &rarr; Inverted String: <strong>"${revStr}"</strong>`;
      }
    }
