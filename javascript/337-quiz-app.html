const questions = [
      {
        q: "Which HTML5 element is used to define navigation links?",
        options: ["<navigation>", "<navigate>", "<nav>", "<links>"],
        answer: 2
      },
      {
        q: "Which CSS property is used to create a 2-dimensional grid layout?",
        options: ["display: flex", "display: grid", "display: block", "float: left"],
        answer: 1
      },
      {
        q: "Which method prevents an event from propagating up the DOM tree in JavaScript?",
        options: ["event.preventDefault()", "event.stopPropagation()", "event.stop()", "event.halt()"],
        answer: 1
      },
      {
        q: "What is the typical storage capacity limit for HTML5 localStorage?",
        options: ["4 KB", "5 MB to 10 MB", "1 GB", "Unlimited"],
        answer: 1
      }
    ];

    let currentQ = 0;
    let score = 0;

    const qNum = document.getElementById("questionNum");
    const qText = document.getElementById("questionText");
    const optBox = document.getElementById("optionsBox");
    const nextBtn = document.getElementById("btnNextQ");
    const progBar = document.getElementById("quizProgressBar");
    const scoreBadge = document.getElementById("scoreBadge");
    const quizContainer = document.getElementById("quizContainer");

    function renderQuestion() {
      const q = questions[currentQ];
      qNum.textContent = `Question ${currentQ + 1} of ${questions.length}`;
      qText.textContent = q.q;
      nextBtn.disabled = true;

      const pct = ((currentQ + 1) / questions.length) * 100;
      progBar.style.width = `${pct}%`;

      optBox.innerHTML = "";
      q.options.forEach((opt, idx) => {
        const btn = document.createElement("button");
        btn.className = "quiz-option-btn";
        btn.textContent = opt;
        btn.onclick = () => selectAnswer(idx, btn);
        optBox.appendChild(btn);
      });
    }

    function selectAnswer(idx, btn) {
      const q = questions[currentQ];
      const allBtns = optBox.querySelectorAll(".quiz-option-btn");
      allBtns.forEach(b => b.disabled = true);

      if (idx === q.answer) {
        btn.classList.add("correct");
        score++;
      } else {
        btn.classList.add("wrong");
        allBtns[q.answer].classList.add("correct");
      }

      scoreBadge.textContent = `Score: ${score} / ${questions.length}`;
      nextBtn.disabled = false;
    }

    function nextQuestion() {
      currentQ++;
      if (currentQ < questions.length) {
        renderQuestion();
      } else {
        showResults();
      }
    }

    function showResults() {
      const pct = Math.round((score / questions.length) * 100);
      quizContainer.innerHTML = `
        <div style="text-align: center; padding: 2rem 1rem;">
          <div style="font-size: 4rem; margin-bottom: 0.5rem;">🏆</div>
          <h2 style="color: #38bdf8; margin-bottom: 0.5rem;">Quiz Examination Completed!</h2>
          <p style="color: #cbd5e1; margin-bottom: 1.5rem;">Candidate: <strong>Shaik Roshan</strong></p>
          
          <div style="font-size: 3rem; font-family: 'Fira Code', monospace; font-weight: 800; color: #10b981; margin-bottom: 0.5rem;">
            ${score} / ${questions.length} (${pct}%)
          </div>

          <p style="color: #94a3b8; margin-bottom: 2rem;">
            ${pct >= 75 ? 'Outstanding mastery in HTML, CSS & JavaScript!' : 'Good attempt! Review questions to improve.'}
          </p>

          <button class="btn btn-primary" onclick="restartQuiz()">🔄 Retake Quiz</button>
        </div>
      `;
    }

    function restartQuiz() {
      currentQ = 0;
      score = 0;
      scoreBadge.textContent = `Score: 0 / ${questions.length}`;
      location.reload();
    }

    renderQuestion();
