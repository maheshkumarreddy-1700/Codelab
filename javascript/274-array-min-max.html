Array min max

Question: Write a JavaScript program to array min max.
▶ Live Output

JavaScript Code
function scanMinMax() {
      const raw = document.getElementById('arr-input').value;
      const res = document.getElementById('res-minmax');
      const arr = raw.split(',').map(item => parseFloat(item.trim())).filter(n => !isNaN(n));

      if (!arr.length) {
        res.textContent = "Please enter valid comma-separated numbers.";
        return;
      }

      let min = arr[0], minIdx = 0;
      let max = arr[0], maxIdx = 0;

      for (let i = 1; i < arr.length; i++) {
        if (arr[i] < min) { min = arr[i]; minIdx = i; }
        if (arr[i] > max) { max = arr[i]; maxIdx = i; }
      }

      res.innerHTML = `
        Parsed Elements (${arr.length} items): [ ${arr.join(', ')} ]<br><br>
        <span style="color: #ef4444; font-weight: 700;">Smallest Element (Min):</span> <strong>${min}</strong> (Found at Index [${minIdx}])<br>
        <span style="color: #16a34a; font-weight: 700;">Largest Element (Max):</span> <strong>${max}</strong> (Found at Index [${maxIdx}])
      `;
    }
    scanMinMax();
