Form events

Question: Write a JavaScript program to form events.
▶ Live Output
Working example
JavaScript Code
const stream = document.getElementById("eventStream");
    const nameInp = document.getElementById("fieldName");
    const branchSel = document.getElementById("fieldBranch");
    const notesInp = document.getElementById("fieldNotes");

    function logStream(type, desc) {
      const row = document.createElement("div");
      row.className = "stream-item";
      row.innerHTML = `<span class="tag-${type}">[${type.toUpperCase()}]</span><span>${desc}</span>`;
      stream.prepend(row);
    }

    function clearStream() {
      stream.innerHTML = "";
    }

    [nameInp, notesInp].forEach(el => {
      el.addEventListener("focus", () => logStream("focus", `Focus gained on #${el.id}`));
      el.addEventListener("blur", () => logStream("blur", `Blur (lost focus) on #${el.id}`));
      el.addEventListener("input", (e) => logStream("input", `Input: "${e.target.value}"`));
    });

    branchSel.addEventListener("change", (e) => {
      logStream("change", `Branch selected: "${e.target.value}"`);
    });

    function handleFormSubmit(e) {
      e.preventDefault();
      logStream("submit", "e.preventDefault() called; Simulated submission complete!");
    }
