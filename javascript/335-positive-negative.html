function checkSign() {
      const val = parseFloat(document.getElementById('sign-val').value);
      const res = document.getElementById('res-sign');
      if (isNaN(val)) {
        res.textContent = "Please enter a valid number.";
        return;
      }
      if (val > 0) {
        res.style.color = "#16a34a";
        res.textContent = `${val} is a POSITIVE number (> 0). Math.sign = +1`;
      } else if (val < 0) {
        res.style.color = "#ef4444";
        res.textContent = `${val} is a NEGATIVE number (< 0). Math.sign = -1`;
      } else {
        res.style.color = "#3b82f6";
        res.textContent = `The number is ZERO (0). Math.sign = 0`;
      }
    }
