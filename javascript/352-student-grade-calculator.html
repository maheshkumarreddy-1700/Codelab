Student grade calculator

Question: Write a JavaScript program to student grade calculator.
▶ Live Output
Working example
JavaScript Code
function computeGrades() {
      const inputs = document.querySelectorAll(".mark-inp");
      let total = 0;
      let hasFailed = false;

      inputs.forEach(inp => {
        let val = parseFloat(inp.value) || 0;
        val = Math.max(0, Math.min(100, val));
        total += val;
        if (val < 40) hasFailed = true;
      });

      const percent = (total / (inputs.length * 100)) * 100;
      const gpa = (percent / 9.5).toFixed(1);

      document.getElementById("totMarks").textContent = total;
      document.getElementById("percentVal").textContent = `${percent.toFixed(1)}%`;
      document.getElementById("gpaVal").textContent = gpa;

      const gradeEl = document.getElementById("letterGrade");
      const statusBox = document.getElementById("finalStatus");

      let grade = "F";
      if (hasFailed) {
        grade = "F";
        statusBox.textContent = "RESULT: FAILED IN ONE OR MORE SUBJECTS";
        statusBox.style.background = "#7f1d1d";
        statusBox.style.color = "#fca5a5";
      } else if (percent >= 90) {
        grade = "O";
        statusBox.textContent = "PASSED WITH OUTSTANDING DISTINCTION";
        statusBox.style.background = "#065f46";
        statusBox.style.color = "#34d399";
      } else if (percent >= 80) {
        grade = "A+";
        statusBox.textContent = "PASSED WITH FIRST CLASS DISTINCTION";
        statusBox.style.background = "#065f46";
        statusBox.style.color = "#34d399";
      } else if (percent >= 60) {
        grade = "B+";
        statusBox.textContent = "PASSED IN FIRST CLASS";
        statusBox.style.background = "#1e3a5f";
        statusBox.style.color = "#38bdf8";
      } else {
        grade = "C";
        statusBox.textContent = "PASSED IN PASS CLASS";
        statusBox.style.background = "#78350f";
        statusBox.style.color = "#fcd34d";
      }

      gradeEl.textContent = grade;
    }

    function resetMarks() {
      document.querySelectorAll(".mark-inp").forEach(inp => inp.value = 85);
      computeGrades();
    }

    computeGrades();
