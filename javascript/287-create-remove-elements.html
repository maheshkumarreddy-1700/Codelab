Create remove elements

Question: Write a JavaScript program to create remove elements.
▶ Live Output
Working example
JavaScript Code
const container = document.getElementById("nodesContainer");
    const input = document.getElementById("nodeTextInput");
    let autoIndex = 4;

    function updateCount() {
      const count = container.children.length;
      document.getElementById("nodeCount").textContent = count + " item" + (count === 1 ? "" : "s");
    }

    function createNode(title) {
      const text = title || `Node #${autoIndex++}: Service Worker`;
      const node = document.createElement("div");
      node.className = "node-item";
      
      const span = document.createElement("span");
      span.textContent = text;
      node.appendChild(span);

      const delBtn = document.createElement("button");
      delBtn.className = "del-btn";
      delBtn.innerHTML = "&times;";
      delBtn.onclick = () => {
        node.remove();
        updateCount();
      };
      node.appendChild(delBtn);

      return node;
    }

    function appendNode() {
      const val = input.value.trim();
      const node = createNode(val);
      container.appendChild(node);
      input.value = "";
      updateCount();
    }

    function prependNode() {
      const val = input.value.trim();
      const node = createNode(val);
      container.prepend(node);
      input.value = "";
      updateCount();
    }

    function clearAllNodes() {
      container.innerHTML = "";
      updateCount();
    }

    updateCount();
