const weatherData = {
      "hyderabad": { city: "Hyderabad, IN", temp: 29, cond: "Partly Cloudy", icon: "⛅", hum: "62%", wind: "14 km/h", uv: "Moderate (5)" },
      "london": { city: "London, UK", temp: 15, cond: "Light Rain Showers", icon: "🌧️", hum: "84%", wind: "22 km/h", uv: "Low (2)" },
      "tokyo": { city: "Tokyo, JP", temp: 22, cond: "Clear Sunny Sky", icon: "☀️", hum: "48%", wind: "9 km/h", uv: "High (7)" },
      "new york": { city: "New York, US", temp: 19, cond: "Mild & Breezy", icon: "💨", hum: "54%", wind: "18 km/h", uv: "Moderate (4)" },
      "dubai": { city: "Dubai, UAE", temp: 38, cond: "Hot & Clear", icon: "☀️", hum: "35%", wind: "12 km/h", uv: "Extreme (10)" },
      "paris": { city: "Paris, FR", temp: 18, cond: "Overcast", icon: "☁️", hum: "68%", wind: "15 km/h", uv: "Moderate (3)" }
    };

    function searchWeather() {
      const input = document.getElementById("cityInput");
      const city = input.value.toLowerCase().trim();
      if (!city) return;

      let result = weatherData[city];

      if (!result) {
        // Procedural simulation for unlisted cities
        const t = Math.floor(Math.random() * 20 + 14);
        result = {
          city: input.value.toUpperCase(),
          temp: t,
          cond: "Sunny Intervals",
          icon: "🌤️",
          hum: `${Math.floor(Math.random() * 30 + 40)}%`,
          wind: `${Math.floor(Math.random() * 15 + 8)} km/h`,
          uv: "Moderate (4)"
        };
      }

      document.getElementById("cityName").textContent = result.city;
      document.getElementById("weatherIcon").textContent = result.icon;
      document.getElementById("weatherCondition").textContent = result.cond;
      document.getElementById("tempDisplay").textContent = `${result.temp}°C`;
      document.getElementById("valHumidity").textContent = result.hum;
      document.getElementById("valWind").textContent = result.wind;
      document.getElementById("valUv").textContent = result.uv;
    }
