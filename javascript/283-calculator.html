Calculator

Question: Write a JavaScript program to calculator.

let currentInput = "0";
    let previousInput = "";
    let operation = null;

    const screen = document.getElementById("calcScreen");
    const history = document.getElementById("calcHistory");

    function updateScreen() {
      screen.textContent = currentInput;
      if (operation) {
        history.textContent = `${previousInput} ${operation}`;
      } else {
        history.textContent = "";
      }
    }

    function appendNum(num) {
      if (currentInput === "0" || currentInput === "Error") {
        currentInput = num;
      } else {
        currentInput += num;
      }
      updateScreen();
    }

    function appendDot() {
      if (!currentInput.includes(".")) {
        currentInput += ".";
        updateScreen();
      }
    }

    function appendOperator(op) {
      if (operation !== null && currentInput !== "0") {
        calculateResult();
      }
      previousInput = currentInput;
      operation = op;
      currentInput = "0";
      updateScreen();
    }

    function deleteChar() {
      if (currentInput.length > 1) {
        currentInput = currentInput.slice(0, -1);
      } else {
        currentInput = "0";
      }
      updateScreen();
    }

    function clearAll() {
      currentInput = "0";
      previousInput = "";
      operation = null;
      updateScreen();
    }

    function calculateResult() {
      if (operation === null || previousInput === "") return;

      const prev = parseFloat(previousInput);
      const curr = parseFloat(currentInput);
      let res = 0;

      switch (operation) {
        case '+': res = prev + curr; break;
        case '-': res = prev - curr; break;
        case '*': res = prev * curr; break;
        case '/': res = (curr === 0) ? "Error" : prev / curr; break;
        case '%': res = (prev * curr) / 100; break;
      }

      history.textContent = `${previousInput} ${operation} ${currentInput} =`;
      currentInput = typeof res === "number" ? String(Math.round(res * 100000) / 100000) : res;
      operation = null;
      previousInput = "";
      screen.textContent = currentInput;
    }

    // Keyboard support
    window.addEventListener("keydown", (e) => {
      if (e.key >= "0" && e.key <= "9") appendNum(e.key);
      else if (e.key === ".") appendDot();
      else if (["+", "-", "*", "/"].includes(e.key)) appendOperator(e.key);
      else if (e.key === "Enter" || e.key === "=") calculateResult();
      else if (e.key === "Backspace") deleteChar();
      else if (e.key === "Escape") clearAll();
    });
